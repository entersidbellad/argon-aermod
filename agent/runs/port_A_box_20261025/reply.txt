"""AERMOD 26135 Python port: package entry point."""

from __future__ import annotations

import numpy as xp
import glue

from .state import AermodState
from .hrloop import hrloop


def run_case(
    project: glue.Project,
    met: glue.Met,
    setup: dict,
    *,
    debug: bool = False,
) -> glue.Result:
    """Run HRLOOP and all in-scope AERMOD physics for a prepared case."""
    state = AermodState.from_setup(project, met, setup, debug=debug)
    hrloop(state)
    dates = xp.asarray(state.dates_out, dtype=xp.int64)
    if len(state.conc_out) > 0:
        conc = xp.stack(state.conc_out, axis=0).astype(xp.float64)
    else:
        conc = xp.zeros((0, state.NUMREC), dtype=xp.float64)
    return glue.Result(dates=dates, conc=conc, debug=state.debug_rows)

def f_nint(x: float) -> int:
    """Fortran NINT / IDNINT: round half away from zero."""
    return int(math.copysign(math.floor(abs(x) + 0.5), x))


def f_int(x: float) -> int:
    """Fortran INT / IDINT: truncate toward zero."""
    return int(math.trunc(x))


def f_idiv(a: int, b: int) -> int:
    """Fortran integer division: truncate toward zero."""
    return int(math.trunc(a / b))


def f_mod_int(a: int, b: int) -> int:
    """Fortran integer MOD(A, B): sign of A."""
    return a - f_idiv(a, b) * b


def f_mod_real(a: float, b: float) -> float:
    """Fortran real MOD(A, B): math.fmod."""
    return math.fmod(a, b)


def f_sign(a: float, b: float) -> float:
    """Fortran DSIGN(A, B): magnitude of A with sign of B."""
    return math.copysign(abs(a), b)

# Lookup tables for JULIAN (aermod.f:1563-1564)
_NDAY = (0, 31, 59, 90, 120, 151, 181, 212, 243, 273, 304, 334)
_IDYMAX = (31, 29, 31, 30, 31, 30, 31, 31, 30, 31, 30, 31)


def cent_date(iyear: int, istrt_wind: int = 50, istrt_cent: int = 19) -> tuple[int, int]:
    """SUBROUTINE CENT_DATE (meset.f): convert 2-digit or 4-digit year to (2-digit, 4-digit)."""
    if istrt_wind <= iyear <= 99:
        iyr = istrt_cent * 100 + iyear
    elif iyear < istrt_wind:
        iyr = (istrt_cent + 1) * 100 + iyear
    else:
        iyr = iyear
        iyear = iyr - 100 * f_idiv(iyr, 100)
    return iyear, iyr


def julian(state: AermodState, inyr: int, inmn: int, indy: int) -> int:
    """SUBROUTINE JULIAN (aermod.f)."""
    jdy = 0

    # Check for Invalid Month or Day
    if inmn < 1 or inmn > 12:
        state.RUNERR = True
        raise RuntimeError(f"AERMOD fatal error 203 in JULIAN: MONTH = {inmn:2d}")
    # Fortran IDYMAX(INMN) -> _IDYMAX[inmn - 1]
    elif indy > _IDYMAX[inmn - 1]:
        state.RUNERR = True
        raise RuntimeError(f"AERMOD fatal error 203 in JULIAN: DAY={indy:2d} MO={inmn:2d}")

    # Determine JULIAN Day Number; For Non-Leap Year First
    if (f_mod_int(inyr, 4) != 0) or (
        f_mod_int(inyr, 100) == 0 and f_mod_int(inyr, 400) != 0
    ):
        # Not a Leap Year
        if inmn != 2 or (inmn == 2 and indy <= 28):
            # Fortran NDAY(INMN) -> _NDAY[inmn - 1]
            jdy = indy + _NDAY[inmn - 1]
        else:
            jdy = 60
            state.RUNERR = True
            raise RuntimeError(f"AERMOD fatal error 370 in JULIAN: YR= {inyr:4d}")
    else:
        # Leap Year
        # Fortran NDAY(INMN) -> _NDAY[inmn - 1]
        jdy = indy + _NDAY[inmn - 1]
        if inmn > 2:
            jdy = jdy + 1

    return jdy


def hrloop(state: AermodState) -> None:
    """SUBROUTINE HRLOOP (aermod.f)."""
    state.EOF = False
    if state.L_MorningTrans is not None:
        state.L_MorningTrans[:] = False
    state.KURDAT = 0
    state.KURPFL = 0
    state.FULLDATE = 0

    # Begin Hourly LOOP
    while state.FULLDATE <= state.IEDATE and not state.EOF:
        # Retrieve One Hour of Meteorology --- CALL METEXT
        metext(state)

        # Check for runtime error generated in call to METEXT
        if state.RUNERR:
            break

        # Fortran IPROC(JDAY) / IPROCL(JDAY) -> state.IPROC[state.JDAY - 1]
        jday_idx = state.JDAY - 1
        in_day_range = (
            (not state.L_LeapYear and state.IPROC[jday_idx] == 1)
            or (state.L_LeapYear and state.IPROCL[jday_idx] == 1)
        ) if 1 <= state.JDAY <= 366 else False

        if (
            state.FULLDATE >= state.ISDATE
            and state.FULLDATE <= state.IEDATE
            and in_day_range
            and not state.EOF
        ):
            if not state.L_SkipMessages:
                state.NTOTHRS = state.NTOTHRS + 1
        elif (
            state.FULLDATE < state.IEDATE
            and state.IEDATE < 2147123124
            and state.EOF
        ):
            # End of met data file(s) reached before user-specified End Date
            raise RuntimeError("AERMOD fatal error 580 in HRLOOP: MET-DATA")
        elif state.EOF or state.FULLDATE > state.IEDATE:
            # End of File or data period has been reached; EXIT hour loop
            break

        ilsave = state.ILINE

        if state.HOURLY:
            raise RuntimeError("HOURLY emissions out of scope")

        if state.RUNERR:
            break

        ilsave = state.ILINE

        if state.L_BACKGRND:
            if state.L_BGHourly:
                raise RuntimeError("Hourly BACKGRND out of scope")
            else:
                if state.IBGSECT > 0:
                    # Fortran L_BGValues(IBGSECT) -> state.L_BGValues[state.IBGSECT - 1]
                    if state.L_BGValues[state.IBGSECT - 1]:
                        raise RuntimeError("BGVAL out of scope")
                    else:
                        state.BGCONC = 0.0
                else:
                    state.BGCONC = 0.0

        if state.RUNERR:
            break

        state.ILINE = ilsave

        if state.PVMRM or state.OLM or state.RUNTTRM or state.GRSM:
            if state.L_O3Hourly:
                raise RuntimeError("Hourly O3 out of scope")
            elif state.IO3SECT > 0:
                # Fortran IO3SECT -> state.IO3SECT - 1
                if state.L_O3VALUES[state.IO3SECT - 1]:
                    raise RuntimeError("OZONVALS out of scope")
                elif state.L_O3VAL[state.IO3SECT - 1]:
                    state.O3CONC = float(state.O3BACK[state.IO3SECT - 1])
                else:
                    state.O3CONC = 0.0
            else:
                state.O3CONC = 0.0

        if state.GRSM:
            if state.L_NOxHourly:
                raise RuntimeError("Hourly NOx out of scope")
            elif state.INOXSECT > 0:
                # Fortran INOXSECT -> state.INOXSECT - 1
                if state.L_NOX_VALS[state.INOXSECT - 1]:
                    raise RuntimeError("VARYNOXVALS out of scope")
                elif state.L_NOXVALUE[state.INOXSECT - 1]:
                    state.NOXBGCONC = float(state.NOXBACK[state.INOXSECT - 1])
                else:
                    state.NOXBGCONC = 0.0
            else:
                state.NOXBGCONC = 0.0

        if state.RUNERR:
            break

        state.ILINE = ilsave

        if state.SCIM and not state.EOF:
            state.SCIMHR = False
            state.NSKIPTOT = state.NSKIPTOT + 1
            if state.ILINE <= 24 and state.IHOUR == state.NREGSTART:
                state.IFIRSTHR = state.ILINE
                state.SCIMHR = True
            elif (
                state.ILINE > state.NREGSTART
                and f_mod_int(state.ILINE - state.IFIRSTHR, state.NREGINT) == 0
            ):
                state.SCIMHR = True
            else:
                continue

        if (
            (
                (not state.RSTINP and state.FULLDATE >= state.ISDATE)
                or (state.RSTINP and not state.MULTYR and state.FULLDATE > state.ISDATE)
                or (state.MULTYR and state.FULLDATE >= state.ISDATE)
            )
            and state.FULLDATE <= state.IEDATE
            and in_day_range
            and not state.EOF
            and not state.RUNERR
        ):
            state.hr_conc[:] = 0.0

            if state.CLMHR and state.CLMPRO:
                # Check for Calm Hr & Processing and Increment Counters
                # Fortran IAVE = 1..NUMAVE -> iave in range(state.NUMAVE)
                for iave in range(state.NUMAVE):
                    state.NUMHRS[iave] = state.NUMHRS[iave] + 1
                    state.NUMCLM[iave] = state.NUMCLM[iave] + 1
                if state.PERIOD or state.ANNUAL:
                    if not state.SCIM or (state.SCIM and state.SCIMHR):
                        state.IANHRS = state.IANHRS + 1
                        state.IANCLM = state.IANCLM + 1
                if state.SEASONHR:
                    # Fortran NSEAHR(ISEAS, IHOUR) -> state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1]
                    state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1] += 1
                    state.NSEACM[state.ISEAS - 1, state.IHOUR - 1] += 1
            elif state.MSGHR and state.MSGPRO:
                # Check for Missing Hour & Processing and Increment Counters
                # Fortran IAVE = 1..NUMAVE -> iave in range(state.NUMAVE)
                for iave in range(state.NUMAVE):
                    state.NUMHRS[iave] = state.NUMHRS[iave] + 1
                    state.NUMMSG[iave] = state.NUMMSG[iave] + 1
                if state.PERIOD or state.ANNUAL:
                    if not state.SCIM or (state.SCIM and state.SCIMHR):
                        state.IANHRS = state.IANHRS + 1
                        state.IANMSG = state.IANMSG + 1
                if state.SEASONHR:
                    # Fortran NSEAHR(ISEAS, IHOUR) -> state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1]
                    state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1] += 1
                    state.NSEACM[state.ISEAS - 1, state.IHOUR - 1] += 1
            elif state.ZI <= 0.0:
                # Write Out The Informational Message & Increment Counters
                # Fortran IAVE = 1..NUMAVE -> iave in range(state.NUMAVE)
                for iave in range(state.NUMAVE):
                    state.NUMHRS[iave] = state.NUMHRS[iave] + 1
                if state.PERIOD or state.ANNUAL:
                    if not state.SCIM or (state.SCIM and state.SCIMHR):
                        state.IANHRS = state.IANHRS + 1
                if state.SEASONHR:
                    # Fortran NSEAHR(ISEAS, IHOUR) -> state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1]
                    state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1] += 1
            else:
                # Set CALCS Flag, Increment Counters & Calculate HRVAL
                state.CALCS = True
                # Fortran IAVE = 1..NUMAVE -> iave in range(state.NUMAVE)
                for iave in range(state.NUMAVE):
                    state.NUMHRS[iave] = state.NUMHRS[iave] + 1
                if state.PERIOD or state.ANNUAL:
                    if not state.SCIM or (state.SCIM and state.SCIMHR):
                        state.IANHRS = state.IANHRS + 1
                if state.SEASONHR:
                    # Fortran NSEAHR(ISEAS, IHOUR) -> state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1]
                    state.NSEAHR[state.ISEAS - 1, state.IHOUR - 1] += 1

                # Calculate CONC or DEPOS Values --- CALL CALC
                calc(state)

            # Record 1-hour POSTFILE concentration for this processed hour
            state.dates_out.append(int(state.KURDAT))
            state.conc_out.append(state.hr_conc.copy())

            # Begin Averaging Period LOOP
            # Fortran IAVE = 1..NUMAVE -> iave in range(state.NUMAVE)
            for iave in range(state.NUMAVE):
                state.IAVE = iave + 1
                kave_val = int(state.KAVE[iave])
                if f_mod_int(state.IHOUR, kave_val) == 0 or (

def _read_sfc_record(state: AermodState, idx: int) -> int:
    """Read surface record idx (0-based) into state; return IJDAY."""
    s = state.met.surface
    state.IYEAR = int(s["IYEAR"][idx])
    state.IMONTH = int(s["IMONTH"][idx])
    state.IDAY = int(s["IDAY"][idx])

    state.WDREF = float(s["WDREF"][idx])

def metext(state: AermodState) -> None:
    """SUBROUTINE METEXT (metext.f)."""
    c1 = 5.31e-13
    c2 = 60.0
    c3 = 1.12
    stefb = 5.67e-08

    state.PATH = "MX"

    # Save Value of Last YR/MN/DY/HR and Previous Hour
    state.IPDATE = state.KURDAT
    state.IPYEAR = state.IYR
    state.IPHOUR = state.IHOUR

    # Initialize USTAR, OBULEN, SFCZ0, QSW, IPCODE, AND PRATE to ZERO for hour
    state.USTAR = 0.0
    state.OBULEN = 0.0
    state.SFCZ0 = 0.0
    state.QSW = 0.0
    state.IPCODE = 0
    state.PRATE = 0.0

    state.RESET_SA = False
    state.RESET_SW = False

    # Increment line counter for messages related to met data file
    state.ILINE = state.ILINE + 1
    state.DUMMY = "SURFFILE"

    # Check for EOF on surface file
    if state.sfc_cursor >= len(state.met):
        state.EOF = True
        if state.ILINE == 1:
            state.RUNERR = True
            raise RuntimeError(f"AERMOD fatal error 580 in METEXT: {state.DUMMY}")
        return

    h_idx = state.sfc_cursor
    state.sfc_cursor += 1
    ijday = _read_sfc_record(state, h_idx)

    if state.LDPART or state.LWPART or state.LDGAS or state.LWGAS or state.GRSM:
        if (
            state.OBULEN > 0.0
            or state.OBULEN < -99990.0
            or state.TA < 0.0
            or state.ALBEDO == 1.0
            or state.BOWEN == 0.0
        ):
            state.QSW = 0.0
        else:
            rn = (1.0 + 1.0 / state.BOWEN) * state.SFCHF / 0.9
            state.QSW = (
                rn * (1.0 + c3)
                - c1 * state.TA**6
                + stefb * state.TA**4
                - c2 * 0.1 * float(state.NCLOUD)
            ) / (1.0 - state.ALBEDO)

        if state.LDPART or state.LDGAS:
            if state.TA < 0.0 or state.PRATE < 0.0:
                state.Wnew = state.Wold
            else:
                state.EsTa = 0.6112 * math.exp(19.83 - 5417.4 / state.TA)
                es25 = 3.167
                state.Wnew = state.Wold + state.PREC1 - 0.5 * state.f2 * state.EsTa / es25
                state.Wold = state.Wnew
                state.f2 = state.Wnew / 200.0
                if state.f2 <= 0.01:
                    state.f2 = 0.01
                if state.f2 > 1.0:
                    state.f2 = 1.0

    # Set the stability logical variables, which are needed in COMPTG
    if state.OBULEN > 0.0:
        state.UNSTAB = False
        state.STABLE = True
    else:
        state.UNSTAB = True
        state.STABLE = False

    # Initialize the profile data to missing; READ profile data
    pflini(state)
    level = 1
    jflag = 0
    state.DUMMY = "PROFFILE"
    state.L_TurbData = False

    p_start = int(state.met.profile_start[h_idx])
    p_end = int(state.met.profile_start[h_idx + 1])
    p_row = p_start
    p = state.met.profile

    while jflag == 0:
        if p_row >= p_end:
            state.EOF = True
            if state.ILINE == 1:
                state.RUNERR = True
                raise RuntimeError(f"AERMOD fatal error 580 in METEXT: {state.DUMMY}")
            return

        state.KYEAR = int(p["KYEAR"][p_row])
        state.KMONTH = int(p["KMONTH"][p_row])
        state.KDAY = int(p["KDAY"][p_row])
        state.KHOUR = int(p["KHOUR"][p_row])
        # Fortran LEVEL = 1..MXPLVL -> level - 1 in 0-based arrays
        lvl_idx = level - 1
        state.PFLHT[lvl_idx] = float(p["PFLHT"][p_row])
        jflag = int(p["JFLAG"][p_row])
        state.PFLWD[lvl_idx] = float(p["PFLWD"][p_row])
        state.PFLWS[lvl_idx] = float(p["PFLWS"][p_row])
        state.PFLTA[lvl_idx] = float(p["PFLTA"][p_row])
        state.PFLSA[lvl_idx] = float(p["PFLSA"][p_row])
        state.PFLSW[lvl_idx] = float(p["PFLSW"][p_row])
        p_row += 1

        # Convert the data to the required units
        pflcnv(state, level)

        if (0.0 < state.PFLSA[lvl_idx] < 99.0) or (
            0.0 < state.PFLSW[lvl_idx] < 99.0
        ):
            state.L_TurbData = True

        state.NPLVLS = level
        state.IFLAG[lvl_idx] = jflag

        if jflag == 0:
            level = level + 1
            if level > state.MXPLVL:
                if not state.PFLERR:
                    state.PFLERR = True
                    state.RUNERR = True
                    raise RuntimeError(
                        f"AERMOD fatal error 465 in METEXT: MXPLVL = {state.MXPLVL:8d}"
                    )
                level = state.MXPLVL

        # Fortran LEVEL -> level - 1
        lvl_idx2 = level - 1
        if 0.0 < state.PFLSA[lvl_idx2] < 99.0:
            state.L_TurbData = True
            state.L_Got_SigA = True
        if 0.0 < state.PFLSW[lvl_idx2] < 99.0:
            state.L_TurbData = True
            state.L_Got_SigW = True

    if state.DFAULT and state.L_TurbData and state.L_AdjUstar:
        state.RUNERR = True
        raise RuntimeError("AERMOD fatal error 401 in METEXT: TurbData with ADJ_U*")

    if state.L_TurbData:
        if state.L_AdjUstar and state.DFAULT:
            state.RUNERR = True
            raise RuntimeError("AERMOD fatal error 401 in METEXT: TurbData with ADJ_U*")

    # Compute the vertical potential temperature gradient profile
    if not state.RUNERR:
        state.NTGLVL = 0
        comptg(state)

    if state.ILINE == 1:
        state.JDAY = ijday
        if not state.L_SkipMessages:
            metdat(state, state.IOUNIT)
        if state.SUMMFILE:
            if not state.L_SkipMessages:
                metdat(state, state.ISUMUNT)

    # Set Meteorological Variables for Current Hour
    set_metdata(state)

    if state.ILINE == 1:
        if state.IHOUR > 1 and (state.PM25AVE or state.NO2AVE or state.SO2AVE):
            if state.L_MAXDCONT:
                state.RUNERR = True
                raise RuntimeError(
                    f"AERMOD fatal error 491 in METEXT: First Hr= {state.IHOUR:02d}"
                )

        # Fortran IMSTAT(7) -> state.IMSTAT[6]
        if state.IYR != state.ISYEAR and state.IMSTAT[6] == 0:
            state.ISYEAR = state.IYR
        elif state.IYR != state.ISYEAR and state.IMSTAT[6] > 0:
            state.ISYEAR = state.IYR
            state.RUNERR = True
            raise RuntimeError(
                f"AERMOD fatal error 493 in METEXT: StartYR {state.IYR:4d}"
            )

        if (
            state.PM25AVE
            or state.NO2AVE
            or state.SO2AVE
            or state.ANNUAL
            or state.MULTYR
        ):
            # Fortran IMSTAT(6) -> state.IMSTAT[5]
            if state.IMSTAT[5] == 0:
                state.ISYR = state.IYEAR
                state.ISHR = state.IHOUR
                state.ISDY = state.IDAY
                state.ISMN = state.IMONTH

                _, state.ISYR = cent_date(
                    state.IYEAR, state.ISTRT_WIND, state.ISTRT_CENT
                )
                state.ISJDAY = julian(state, state.ISYR, state.ISMN, state.ISDY)

                if state.IHOUR > 1:
                    state.IENDHOUR = state.IHOUR - 1
                    state.IENDDY = state.IDAY
                    state.IENDMN = state.IMONTH
                else:
                    state.IENDHOUR = 24
                    if state.IDAY > 1:
                        state.IENDDY = state.IDAY - 1
                        state.IENDMN = state.IMONTH
                    else:
                        state.IENDMN = state.IMONTH - 1
                        if state.IENDMN == 0:
                            state.IENDMN = 12
                        # Fortran IDYMAX(IENDMN) -> _IDYMAX[state.IENDMN - 1]
                        state.IENDDY = _IDYMAX[state.IENDMN - 1]
                if not state.RSTINP:
                    state.ISDATE = state.FULLDATE

        # Fortran IMSTAT(6) -> state.IMSTAT[5]
        if (
            not state.MULTYR
            and state.IMSTAT[5] == 1
            and state.FULLDATE > state.ISDATE
        ):
            state.RUNERR = True
            raise RuntimeError(
                f"AERMOD fatal error 483 in METEXT: {state.FULLDATE:010d}"
            )
        elif not state.MULTYR and state.RSTINP and state.FULLDATE > state.ISDATE:
            state.RUNERR = True
            raise RuntimeError("AERMOD fatal error 484 in METEXT")
        elif (
            state.MULTYR
            and state.RSTINP
            and state.IMSTAT[5] == 0
            and state.FULLDATE < state.ISDATE
        ):
            state.RUNERR = True
            raise RuntimeError("AERMOD fatal error 487 in METEXT")


def set_metdata(state: AermodState) -> None:
    """SUBROUTINE SET_METDATA (metext.f)."""
    # Set lower limit of 1.0 meter for mixing heights
    if 0.0 <= state.ZICONV < 1.0:
        state.ZICONV = 1.0
    if 0.0 <= state.ZIMECH < 1.0:
        state.ZIMECH = 1.0

    # Set the date variables for this hour
    set_dates(state)

    if state.MONTH and state.IHOUR == 24:
        # Fortran NDAY(IMONTH) -> _NDAY_END[state.IMONTH - 1]
        if (
            state.IMONTH == 1
            or (f_mod_int(state.IYR, 4) != 0)
            or (f_mod_int(state.IYR, 100) == 0 and f_mod_int(state.IYR, 400) != 0)
        ):
            if state.JDAY == _NDAY_END[state.IMONTH - 1]:
                state.ENDMON = True
        else:
            if state.JDAY == _NDAY_END[state.IMONTH - 1] + 1:
                state.ENDMON = True

    # Check Data for Calms, Missing, Out-of-Range Values --- CALL METCHK
    if not state.L_SkipMessages:
        metchk(state)

    # Limit ZI to 4000 meters.
    if state.ZICONV > 4000.0:
        state.ZICONV = 4000.0
    if state.ZIMECH > 4000.0:
        state.ZIMECH = 4000.0

    # Select appropriate mixing height from convective and mechanical values
    if not state.MSGHR and not state.CLMHR and state.OBULEN < 0.0:
        state.ZI = max(state.ZICONV, state.ZIMECH)
    elif not state.MSGHR and not state.CLMHR:
        state.ZI = state.ZIMECH
    else:
        state.ZI = -999.0

    # Set lower limit of 1.0 meter for mixing height
    if 0.0 <= state.ZI < 1.0:
        state.ZI = 1.0

    # Assign ZI to ZIRUR for URBAN option
    if state.URBAN:
        state.ZIRUR = state.ZI

    # Apply ROTANG Adjustment to Wind Direction
    if abs(state.ROTANG) > 0.0000001:
        state.WDREF = state.WDREF - state.ROTANG
        if state.WDREF <= 0.0:
            state.WDREF = state.WDREF + 360.0
        # Fortran NL = 1..NPLVLS -> nl in range(state.NPLVLS)
        for nl in range(state.NPLVLS):
            if state.PFLWD[nl] > 0.0:
                state.PFLWD[nl] = state.PFLWD[nl] - state.ROTANG
                if state.PFLWD[nl] <= 0.0:
                    state.PFLWD[nl] = state.PFLWD[nl] + 360.0

    if state.L_BLSOURCE:
        state.BLTA = state.TA

    if not state.L_SkipMessages:
        state.URBSTAB = False

    if not state.RUNERR and not state.L_SkipMessages:
        if not state.CLMHR and not state.MSGHR:
            if state.OBULEN > 0.0:
                state.UNSTAB = False
                state.STABLE = True
            else:
                state.UNSTAB = True
                state.STABLE = False

            # Fortran IPROCL(JDAY) / IPROC(JDAY) -> index state.JDAY - 1
            jday_idx = state.JDAY - 1
            in_day = (
                (state.L_LeapYear and state.IPROCL[jday_idx] == 1)
                or (not state.L_LeapYear and state.IPROC[jday_idx] == 1)
            ) if 1 <= state.JDAY <= 366 else False

            if state.FULLDATE >= state.ISDATE and in_day:
                # Initialize the gridded profile arrays
                state.GRIDSV[:] = -99.0
                state.GRIDSW[:] = -99.0
                state.GRIDWS[:] = -99.0
                state.GRIDWD[:] = -99.0
                state.GRIDTG[:] = -99.0
                state.GRIDPT[:] = -99.0
                if state.URBAN:
                    state.GRDSVR[:] = -99.0
                    state.GRDSVU[:, :] = -99.0
                    state.GRDSWR[:] = -99.0
                    state.GRDSWU[:, :] = -99.0
                    state.GRDTGR[:] = -99.0
                    state.GRDTGU[:, :] = -99.0
                    state.GRDPTR[:] = -99.0
                    state.GRDPTU[:, :] = -99.0

                if state.PVMRM or state.GRSM:
                    state.GRIDEPS[:] = -99.0

                # Get the index from the array of gridded heights immediately below ZI
                state.NDX4ZI = locate(state.GRIDHT, 1, state.MXGLVL, state.ZI)

                # Compute THETA_STAR and DTHDZ for the gridded potential temperature gradient
                tginit(state)

                # Profile all variables here except sv and sw; defer sv and sw until u at zi is known
                grdws(state)
                grdwd(state)
                grdptg(state)
                grdpt(state)

                # Compute density profile for PRIME
                grdden(state)

                # Compute the parameter values at ZI
                if state.NDX4ZI < state.MXGLVL:
                    # Fortran NDX4ZI -> state.NDX4ZI - 1, NDX4ZI+1 -> state.NDX4ZI
                    idx = state.NDX4ZI - 1
                    state.UATZI = gintrp(
                        float(state.GRIDHT[idx]),
                        float(state.GRIDWS[idx]),
                        float(state.GRIDHT[idx + 1]),
                        float(state.GRIDWS[idx + 1]),
                        state.ZI,
                    )
                    state.PTATZI = gintrp(
                        float(state.GRIDHT[idx]),
                        float(state.GRIDPT[idx]),
                        float(state.GRIDHT[idx + 1]),
                        float(state.GRIDPT[idx + 1]),
                        state.ZI,
                    )
                else:
                    # Fortran MXGLVL -> state.MXGLVL - 1
                    state.UATZI = float(state.GRIDWS[state.MXGLVL - 1])
                    state.PTATZI = float(state.GRIDPT[state.MXGLVL - 1])

                # Add turbulence variables here
                grdsv(state)

                # Obtain residual turbulence value before calling GRDSW
                numsw = 0
                sumsw = 0.0
                # Fortran I = 1..NPLVLS -> i in range(state.NPLVLS)
                for i in range(state.NPLVLS):
                    if state.PFLHT[i] >= state.ZI and state.PFLSW[i] >= 0.0:
                        numsw = numsw + 1
                        sumsw = sumsw + float(state.PFLSW[i])

                if numsw > 0:
                    state.SWRMAX = sumsw / float(numsw)
                else:
                    state.SWRMAX = 0.02 * state.UATZI

                grdsw(state)

                if state.NDX4ZI < state.MXGLVL:
                    # Fortran NDX4ZI -> state.NDX4ZI - 1, NDX4ZI+1 -> state.NDX4ZI
                    idx = state.NDX4ZI - 1
                    state.SVATZI = gintrp(
                        float(state.GRIDHT[idx]),
                        float(state.GRIDSV[idx]),
                        float(state.GRIDHT[idx + 1]),
                        float(state.GRIDSV[idx + 1]),
                        state.ZI,
                    )
                    state.SWATZI = gintrp(
                        float(state.GRIDHT[idx]),
                        float(state.GRIDSW[idx]),
                        float(state.GRIDHT[idx + 1]),
                        float(state.GRIDSW[idx + 1]),
                        state.ZI,
                    )
                else:
                    # Fortran MXGLVL -> state.MXGLVL - 1
                    state.SVATZI = float(state.GRIDSV[state.MXGLVL - 1])
                    state.SWATZI = float(state.GRIDSW[state.MXGLVL - 1])

                # Compute the overbar (average) quantities from surface up to ZI
                state.SVAVG = ziaver(
                    state.MXGLVL,
                    state.GRIDHT,
                    state.GRIDSV,
                    state.ZI,
                    state.NDX4ZI,
                    state.SVATZI,
                )
                state.SWAVG = ziaver(
                    state.MXGLVL,
                    state.GRIDHT,
                    state.GRIDSW,
                    state.ZI,
                    state.NDX4ZI,
                    state.SWATZI,
                )
                state.UAVG = ziaver(
                    state.MXGLVL,
                    state.GRIDHT,
                    state.GRIDWS,
                    state.ZI,
                    state.NDX4ZI,
                    state.UATZI,
                )

                if state.PVMRM or state.GRSM:
                    grdeps(state)

                if state.URBAN:
                    urbcalc(state)
                    grdurban(state)

    # Assign sector ID for direction-varying background O3 and NOx
    if state.WDREF <= 0.0 or state.WDREF > 360.0:
        state.IO3SECT = 0
        state.INOXSECT = 0
    else:
        fvref = state.WDREF + 180.0
        if fvref > 360.0:
            fvref = fvref - 360.0
        if state.L_O3Sector:
            if fvref < state.O3SECT[0] or fvref >= state.O3SECT[state.NUMO3Sects - 1]:
                state.IO3SECT = state.NUMO3Sects
            else:
                # Fortran I = 1..NUMO3Sects-1 -> i in range(state.NUMO3Sects - 1)
                for i in range(state.NUMO3Sects - 1):
                    if state.O3SECT[i] <= fvref < state.O3SECT[i + 1]:
                        state.IO3SECT = i + 1
                        break
        else:
            state.IO3SECT = 1

        if state.L_NOXSector:
            if (
                fvref < state.NOXSECT[0]
                or fvref >= state.NOXSECT[state.NumNOxSects - 1]
            ):
                state.INOXSECT = state.NumNOxSects
            else:
                # Fortran I = 1..NumNOxSects-1 -> i in range(state.NumNOxSects - 1)
                for i in range(state.NumNOxSects - 1):
                    if state.NOXSECT[i] <= fvref < state.NOXSECT[i + 1]:
                        state.INOXSECT = i + 1
                        break
        else:
            state.INOXSECT = 1

    # Assign sector ID for direction-varying background
    if state.WDREF <= 0.0 or state.WDREF > 360.0:
        state.IBGSECT = 0
    else:
        fvref = state.WDREF + 180.0
        if fvref > 360.0:
            fvref = fvref - 360.0
        if state.L_BGSector:
            if fvref < state.BGSECT[0] or fvref >= state.BGSECT[state.NUMBGSects - 1]:
                state.IBGSECT = state.NUMBGSects
            else:
                # Fortran I = 1..NUMBGSects-1 -> i in range(state.NUMBGSects - 1)
                for i in range(state.NUMBGSects - 1):
                    if state.BGSECT[i] <= fvref < state.BGSECT[i + 1]:
                        state.IBGSECT = i + 1
                        break
        else:
            state.IBGSECT = 1

    # Set Appropriate Wind Speed Category Index
    if state.UREF <= state.UCAT[0]:
        state.IUCAT = 1
    elif state.UREF <= state.UCAT[1]:
        state.IUCAT = 2
    elif state.UREF <= state.UCAT[2]:
        state.IUCAT = 3
    elif state.UREF <= state.UCAT[3]:
        state.IUCAT = 4
    elif state.UREF <= state.UCAT[4]:
        state.IUCAT = 5
    else:
        state.IUCAT = 6

    if state.FASTAREA or state.L_BLSOURCE:
        from .calc1 import ltopg
        state.KST = ltopg(state)
    else:
        state.KST = 4

    if state.MSGHR and not state.L_SkipMessages:
        if not state.MSGPRO:
            state.RUNERR = True
            raise RuntimeError(
                f"AERMOD fatal error 460 in SET_METDATA: {state.FULLDATE:010d}"
            )


def set_dates(state: AermodState) -> None:
    """SUBROUTINE SET_DATES (metext.f)."""
    if not state.L_SkipMessages:
        state.IYEAR, state.IYR = cent_date(
            state.IYEAR, state.ISTRT_WIND, state.ISTRT_CENT
        )

        if (f_mod_int(state.IYR, 4) != 0) or (
            f_mod_int(state.IYR, 100) == 0 and f_mod_int(state.IYR, 400) != 0
        ):
            state.L_LeapYear = False
        else:
            state.L_LeapYear = True

        if state.JDAY >= 1:
            state.JDAY_PREV = state.JDAY
        else:
            state.JDAY_PREV = 0

        state.JDAY = julian(state, state.IYR, state.IMONTH, state.IDAY)

        state.KURDAT = (
            state.IYEAR * 1000000
            + state.IMONTH * 10000
            + state.IDAY * 100
            + state.IHOUR
        )
        if state.IYR >= 2148:
            state.RUNERR = True
            state.FULLDATE = 2147123124
            raise RuntimeError(f"AERMOD fatal error 365 in SET_DATES: YR= {state.IYR:4d}")
        else:
            state.FULLDATE = (
                state.IYR * 1000000
                + state.IMONTH * 10000
                + state.IDAY * 100
                + state.IHOUR
            )

        if state.KYEAR >= 100:
            state.KYEAR = state.KYEAR - 100 * f_idiv(state.KYEAR, 100)
        state.KURPFL = (
            state.KYEAR * 1000000
            + state.KMONTH * 10000
            + state.KDAY * 100
            + state.KHOUR
        )

    # Determine SEASON index
    # Fortran ISEA_NDX(IMONTH) -> _ISEA_NDX[state.IMONTH - 1]
    state.ISEAS = _ISEA_NDX[state.IMONTH - 1]

    if state.L_DayOfWeekOpts:
        ia = f_idiv(14 - state.IMONTH, 12)
        iy = state.IYR - ia
        im = state.IMONTH + 12 * ia - 2
        id_val = f_mod_int(
            state.IDAY
            + iy
            + f_idiv(iy, 4)
            - f_idiv(iy, 100)
            + f_idiv(iy, 400)
            + f_idiv(31 * im, 12),
            7,
        )
        if 1 <= id_val <= 5:
            state.IDAY_OF_WEEK = 1
        elif id_val == 6:
            state.IDAY_OF_WEEK = 2
        elif id_val == 0:
            state.IDAY_OF_WEEK = 3
        if id_val == 0:
            state.IDAY_OF_WEEK7 = 7
        else:
            state.IDAY_OF_WEEK7 = id_val
    else:
        state.IDAY_OF_WEEK = 1
        state.IDAY_OF_WEEK7 = 1

def chkmsg(state: AermodState) -> None:
    """SUBROUTINE CHKMSG (metext.f)."""
    # Wind speed (meters/second)
    if state.UREF >= 90.0 or state.UREF < 0.0:
        state.MSGHR = True
    # Wind direction (degrees from north)
    elif (state.WDREF > 900.0) or (state.WDREF <= -9.0):
        state.MSGHR = True
    # Ambient temperature (kelvins)
    elif (state.TA > 900.0) or (state.TA <= 0.0):
        state.MSGHR = True
    # Monin-Obukhov length (meters)
    elif state.OBULEN < -99990.0:
        state.MSGHR = True
    # Convective Mixing height (meters)
    elif state.OBULEN < 0.0 and ((state.ZICONV > 90000.0) or (state.ZICONV < 0.0)):
        state.MSGHR = True
    # Mechanical Mixing height (meters)
    elif (state.ZIMECH > 90000.0) or (state.ZIMECH < 0.0):
        state.MSGHR = True
    # Surface friction velocity (meters/second)
    elif state.USTAR < 0.0 or state.USTAR >= 9.0:
        state.MSGHR = True
    # Convective velocity scale (meters/second)
    elif state.WSTAR < 0.0 and (state.OBULEN < 0.0 and state.OBULEN > -99990.0):
        state.MSGHR = True


def metqa(state: AermodState) -> None:
    """SUBROUTINE METQA (metext.f)."""
    # Wind direction:
    if state.WDREF == 0.0:
        state.WDREF = 360.0

    # Fortran NL = 1..NPLVLS -> nl in range(state.NPLVLS)
    for nl in range(state.NPLVLS):
        if state.PFLWD[nl] == 0.0:
            state.PFLWD[nl] = 360.0

    if not state.L_SkipMessages:
        if state.UREFHT < 0.001 and not state.CLMHR and not state.MSGHR:
            state.RUNERR = True
            raise RuntimeError(f"AERMOD fatal error 474 in METQA: {state.KURDAT:08d}")

    # Convective velocity (meters/second):
    if state.WSTAR == 0.0:
        state.WSTAR = 0.001

    # Monin-Obukhov length (meters):
    if abs(state.OBULEN) < 1.00:
        if state.OBULEN < 0.0:
            state.OBULEN = -1.0
        elif state.OBULEN > 0.0:
            state.OBULEN = 1.0
        else:
            state.OBULEN = -1.0 * f_sign(1.0, state.SFCHF)

    # Vertical potential temperature gradient above ZI (K/m)
    if (
        state.ZICONV > 0.0
        and state.OBULEN < 0.0
        and state.OBULEN > -99990.0
        and state.VPTGZI < 0.005
    ):
        state.VPTGZI = 0.005

    # Surface roughness length (m):
    if state.SFCZ0 < 0.0001:
        state.SFCZ0 = 0.0001

    # Check for precipitation rate out of range
    if state.PRATE < 0.0 or state.PRATE > 900.0:
        state.PRATE = 0.0
    else:
        state.TOTAL_PRECIP = state.TOTAL_PRECIP + state.PRATE

def pflcnv(state: AermodState, level: int) -> None:
    """SUBROUTINE PFLCNV (metext.f)."""
    # Fortran LEVEL (1-based) -> idx = level - 1 (0-based)
    idx = level - 1

    # RESET PFLSA AND/OR PFLSW BASED ON TURBULENCE OPTIONS
    if state.PFLSA[idx] < 99.0 and (
        state.TURBOPTS[0]
        or (state.TURBOPTS[1] and state.STABLE)
        or (state.TURBOPTS[2] and state.UNSTAB)
        or state.TURBOPTS[3]
        or (state.TURBOPTS[5] and state.STABLE)
        or (state.TURBOPTS[7] and state.UNSTAB)
    ):
        state.PFLSA[idx] = 99.0
        if not state.RESET_SA:
            state.RESET_SA = True

    if state.PFLSW[idx] < 99.0 and (
        state.TURBOPTS[0]
        or (state.TURBOPTS[1] and state.STABLE)
        or (state.TURBOPTS[2] and state.UNSTAB)
        or state.TURBOPTS[4]
        or (state.TURBOPTS[6] and state.STABLE)
        or (state.TURBOPTS[8] and state.UNSTAB)
    ):
        state.PFLSW[idx] = 99.0
        if not state.RESET_SW:
            state.RESET_SW = True

    # Change the missing value indicator for wind speed to -99.0
    if state.PFLWS[idx] < 0.0 or state.PFLWS[idx] > 90.0:
        state.PFLWS[idx] = -99.0

    # Change the wind direction from 0.0 to 360.0 if wind speed is nonzero
    if (0.0 < state.PFLWS[idx] <= 90.0) and state.PFLWD[idx] == 0.0:
        state.PFLWD[idx] = 360.0

    if state.PFLWD[idx] > 900.0 or state.PFLWD[idx] < 0.0:
        state.PFLWD[idx] = -999.0

    if state.PFLWS[idx] == 0.0 and state.PFLWD[idx] == 0.0:
        state.PFLWS[idx] = -99.0
        state.PFLWD[idx] = -999.0

    # Compute sigmaV from nonmissing wind speed and sigmaTHETA
    if state.PFLWS[idx] > 0.0 and 0.0 <= state.PFLSA[idx] < 99.0:
        sigrad = float(state.PFLSA[idx]) * state.DTORAD
        epsil = math.sin(sigrad) * (1.0 - state.GSIGV * sigrad)
        usubv = float(state.PFLWS[idx]) * math.sqrt(1.0 - epsil * epsil)
        state.PFLSV[idx] = sigrad * usubv
        state.PFLSV[idx] = max(state.SVMIN, float(state.PFLSV[idx]))
        state.L_Got_SigA = True
        state.N_Got_SigA = state.N_Got_SigA + 1
    else:
        state.PFLSV[idx] = -99.0

    # Check for sigmaW data
    if 0.0 < state.PFLSW[idx] < 90.0:
        state.L_Got_SigW = True
        state.N_Got_SigW = state.N_Got_SigW + 1

    # Check for turbulence data with ADJ_U
    if state.L_AdjUstar and state.DFAULT:
        if state.L_Got_SigA and state.L_Got_SigW:
            if state.N_Got_SigA == 1 and state.N_Got_SigW == 1:
                state.FATAL = True
                raise RuntimeError("AERMOD fatal error 401 in PFLCNV: SigA & SigW")
        elif state.L_Got_SigA and not state.L_Got_SigW:
            if state.N_Got_SigA == 1:
                state.FATAL = True
                raise RuntimeError("AERMOD fatal error 401 in PFLCNV: SigA Data")
        elif not state.L_Got_SigA and state.L_Got_SigW:
            if state.N_Got_SigW == 1:
                state.FATAL = True
                raise RuntimeError("AERMOD fatal error 401 in PFLCNV: SigW Data")

    # Convert temperature from degrees Celsius to kelvins
    if -90.0 < state.PFLTA[idx] < 90.0:
        state.PFLTA[idx] = state.PFLTA[idx] + state.DCTODK
    else:
        state.PFLTA[idx] = -999.0

    # Change the missing value indicator for sigmaW to -99.0
    if state.PFLSW[idx] < 0.0 or state.PFLSW[idx] > 90.0:
        state.PFLSW[idx] = -99.0
    else:
        state.PFLSW[idx] = max(state.SWMIN, float(state.PFLSW[idx]))

def ziaver(
    nlvls: int,
    hts: xp.ndarray,
    parray: xp.ndarray,
    zi: float,
    ndxblw: int,
    valzi: float,
) -> float:
    """SUBROUTINE ZIAVER (metext.f)."""
    sum_val = 0.0

    # Sum over each layer of the gridded profile (PARRAY) to the level immediately below ZI
    # Fortran I = 2..NDXBLW -> i in range(1, ndxblw)
    for i in range(1, ndxblw):
        sum_val = sum_val + (float(hts[i]) - float(hts[i - 1])) * 0.5 * (
            float(parray[i]) + float(parray[i - 1])
        )

    if ndxblw < nlvls:
        # Fortran HTS(NDXBLW) -> hts[ndxblw - 1]
        sum_val = sum_val + (zi - float(hts[ndxblw - 1])) * 0.5 * (
            valzi + float(parray[ndxblw - 1])
        )
        pblavg = sum_val / zi
    else:
        # Fortran HTS(NLVLS) -> hts[nlvls - 1]
        sum_val = sum_val + (zi - float(hts[nlvls - 1])) * float(parray[nlvls - 1])
        pblavg = sum_val / zi

    return pblavg


def gintrp(
    htbelo: float, vbelow: float, htabov: float, vabove: float, reqdht: float
) -> float:
    """SUBROUTINE GINTRP (metext.f)."""
    value = vbelow + ((reqdht - htbelo) / (htabov - htbelo)) * (vabove - vbelow)
    return value


def urbcalc(state: AermodState) -> None:
    """SUBROUTINE URBCALC (metext.f)."""
    refzi = 400.0
    cp = 1004.0

    state.RURUSTR = state.USTAR
    state.RUROBULEN = state.OBULEN

    # Fortran IURB = 1..NUMURB -> iurb in range(state.NUMURB)
    for iurb in range(state.NUMURB):
        state.IURB = iurb + 1
        deltur = state.DELTRUR * (
            0.1 * math.log(float(state.URBPOP[iurb]) / state.REFPOP) + 1.0
        )

        if state.STABLE:
            state.ZIURB[iurb] = refzi * (float(state.URBPOP[iurb]) / state.REFPOP) ** 0.25
            state.L_MorningTrans[iurb] = False
        elif state.L_UrbanTransition:
            state.ZIURB[iurb] = refzi * (float(state.URBPOP[iurb]) / state.REFPOP) ** 0.25
            if state.ZICONV > state.ZIURB[iurb]:
                state.L_MorningTrans[iurb] = False
                continue
            else:
                state.L_MorningTrans[iurb] = True
        else:
            state.L_MorningTrans[iurb] = False
            continue

        rho = 101325.0 / (287.04 * state.TA)
        urbhf = 0.03 * deltur * state.USTAR * rho * cp
        state.URBWSTR[iurb] = (
            (state.G / state.TA / rho / cp) * urbhf * float(state.ZIURB[iurb])
        ) ** state.THIRD

        ht7z0 = 7.0 * max(float(state.URBZ0[iurb]), state.SFCZ0)
        urbustrsav = (
            float(state.URBWSTR[iurb])
            * math.sqrt(
                1.6 * (ht7z0 / float(state.ZIURB[iurb])) ** (2.0 * state.THIRD)
            )
            / (
                1.3
                * math.sqrt(
                    1.0 - ht7z0 / max(float(state.ZIURB[iurb]), state.ZIMECH)
                )
            )
        )
        state.URBUSTR[iurb] = max(state.USTAR, urbustrsav)

        state.URBOBULEN[iurb] = -(
            (rho * cp * state.TA * float(state.URBUSTR[iurb]) ** 3)
            / (0.4 * state.G * urbhf)
        )

        if state.L_MorningTrans[iurb]:
            state.URBOBULEN[iurb] = max(
                float(state.URBOBULEN[iurb]), state.RUROBULEN
            )
        else:
            state.URBOBULEN[iurb] = max(
                abs(float(state.URBOBULEN[iurb])), state.RUROBULEN
            )


def grdurban(state: AermodState) -> None:
    """SUBROUTINE GRDURBAN (metext.f)."""
    state.GRDSVR[:] = state.GRIDSV[:]
    state.GRDSWR[:] = state.GRIDSW[:]
    state.GRDTGR[:] = state.GRIDTG[:]
    state.GRDPTR[:] = state.GRIDPT[:]

    # Fortran IURB = 1..NUMURB -> iurb in range(state.NUMURB)
    for iurb in range(state.NUMURB):
        state.IURB = iurb + 1
        if not state.STABLE and not state.L_MorningTrans[iurb]:
            continue

        zdcrs = state.AT1PT2 * float(state.ZIURB[iurb])

        # Fortran I = 1..MXGLVL -> i in range(state.MXGLVL)
        for i in range(state.MXGLVL):
            sv2 = 0.35 * float(state.URBWSTR[iurb]) ** 2
            gridht_i = float(state.GRIDHT[i])
            ziurb_u = float(state.ZIURB[iurb])

            if gridht_i <= ziurb_u:
                svurb = math.sqrt(sv2)
            elif ziurb_u < gridht_i <= zdcrs:
                sv2dcr = min(sv2, 0.25)
                val2 = gintrp(ziurb_u, sv2, zdcrs, sv2dcr, gridht_i)
                svurb = math.sqrt(val2)
            else:
                atzi = math.sqrt(sv2)
                svurb = min(atzi, 0.5)

            if gridht_i <= 0.1 * ziurb_u:
                sw2 = (
                    1.6
                    * (gridht_i / ziurb_u) ** (2.0 * state.THIRD)
                    * float(state.URBWSTR[iurb]) ** 2
                )
                swurb = math.sqrt(sw2)
            elif 0.1 * ziurb_u < gridht_i <= ziurb_u:
                swurb = math.sqrt(0.35 * float(state.URBWSTR[iurb]) ** 2)
            else:
                sw2 = (
                    0.35
                    * float(state.URBWSTR[iurb]) ** 2
                    * math.exp(-(6.0 * (gridht_i - ziurb_u) / ziurb_u))
                )
                swurb = math.sqrt(sw2)

            state.GRDSVU[i, iurb] = math.sqrt(
                float(state.GRIDSV[i]) ** 2 + svurb**2
            )
            state.GRDSWU[i, iurb] = math.sqrt(
                float(state.GRIDSW[i]) ** 2 + swurb**2
            )

            if gridht_i <= ziurb_u:
                state.GRDTGU[i, iurb] = 1.0e-5
            else:
                state.GRDTGU[i, iurb] = state.GRIDTG[i]

    grdpturb(state)


def grdpturb(state: AermodState) -> None:
    """SUBROUTINE GRDPTURB (metext.f)."""
    nbelow = locate(state.GRIDHT, 1, state.MXGLVL, state.TREFHT)
    ptref = state.TA + state.GOVRCP * (state.TREFHT + state.ZBASE)

    # Fortran IURB = 1..NUMURB -> iurb in range(state.NUMURB)
    for iurb in range(state.NUMURB):
        state.IURB = iurb + 1
        if not state.STABLE and not state.L_MorningTrans[iurb]:
            continue

        # Fortran NBELOW -> nbelow - 1
        nb_idx = nbelow - 1
        state.GRDPTU[nb_idx, iurb] = ptref - 0.5 * (
            float(state.GRDTGU[nb_idx + 1, iurb])
            + float(state.GRDTGU[nb_idx, iurb])
        ) * (state.TREFHT - float(state.GRIDHT[nb_idx]))

        # Fortran L = NBELOW-1, 1, -1 -> l_idx in range(nbelow - 2, -1, -1)
        for l_idx in range(nbelow - 2, -1, -1):
            state.GRDPTU[l_idx, iurb] = float(
                state.GRDPTU[l_idx + 1, iurb]
            ) - 0.5 * (
                float(state.GRDTGU[l_idx + 1, iurb])
                + float(state.GRDTGU[l_idx, iurb])
            ) * (
                float(state.GRIDHT[l_idx + 1]) - float(state.GRIDHT[l_idx])
            )

        # Fortran L = NBELOW+1, MXGLVL -> l_idx in range(nbelow, state.MXGLVL)
        for l_idx in range(nbelow, state.MXGLVL):
            state.GRDPTU[l_idx, iurb] = float(
                state.GRDPTU[l_idx - 1, iurb]
            ) + 0.5 * (
                float(state.GRDTGU[l_idx, iurb])
                + float(state.GRDTGU[l_idx - 1, iurb])
            ) * (
                float(state.GRIDHT[l_idx]) - float(state.GRIDHT[l_idx - 1])
            )

def locate(parray: xp.ndarray, lvlblw: int, lvlabv: int, value: float) -> int:
    """SUBROUTINE LOCATE (iblval.f): returns 1-based index NDXBLW."""
    jl = lvlblw - 1
    ju = lvlabv + 1

    while (ju - jl) > 1:
        jm = f_idiv(ju + jl, 2)
        # Fortran PARRAY(JM) -> parray[jm - 1]
        if value >= float(parray[jm - 1]):
            jl = jm
        else:
            ju = jm

    ndxblw = min(jl, lvlabv - 1)
    return ndxblw


def anyavg(
    nlvls: int,
    hts: xp.ndarray,
    parray: xp.ndarray,
    zbot: float,
    ndxabv: int,
    ztop: float,
    ndxblw: int,
) -> float:
    """SUBROUTINE ANYAVG (iblval.f)."""
    from .metext import gintrp

    sum_val = 0.0

    # Check for minimum values of ZTOP and ZBOT
    if zbot < 0.5:
        zbot = 0.5
        ndxabv = 2
    if ztop < 0.51:
        ztop = 0.51
        ndxblw = 2

    if ndxblw < ndxabv:
        # Label 300: take interpolated value halfway between ZBOT and ZTOP
        # Fortran NDXABV-1 -> ndxabv - 2, NDXABV -> ndxabv - 1
        valavg = gintrp(
            float(hts[ndxabv - 2]),
            float(parray[ndxabv - 2]),
            float(hts[ndxabv - 1]),
            float(parray[ndxabv - 1]),
            0.5 * (zbot + ztop),
        )
        return valavg

    if ndxblw > ndxabv:
        # Sum using trapezoidal rule over intermediate profile layers
        # Fortran I = NDXABV+1, NDXBLW -> i in range(ndxabv, ndxblw)
        for i in range(ndxabv, ndxblw):
            sum_val = sum_val + (float(hts[i]) - float(hts[i - 1])) * 0.5 * (
                float(parray[i]) + float(parray[i - 1])
            )

    # Label 200: finish summation over partial layers at bottom (first), then top
    if ndxabv > 1:
        # Fortran NDXABV-1 -> ndxabv - 2, NDXABV -> ndxabv - 1
        valbot = gintrp(
            float(hts[ndxabv - 2]),
            float(parray[ndxabv - 2]),
            float(hts[ndxabv - 1]),
            float(parray[ndxabv - 1]),
            zbot,
        )
        sum_val = sum_val + (float(hts[ndxabv - 1]) - zbot) * 0.5 * (
            valbot + float(parray[ndxabv - 1])
        )
    else:
        sum_val = sum_val + (float(hts[0]) - zbot) * float(parray[0])

    if ndxblw < nlvls:
        # Fortran NDXBLW -> ndxblw - 1, NDXBLW+1 -> ndxblw
        valtop = gintrp(
            float(hts[ndxblw - 1]),
            float(parray[ndxblw - 1]),
            float(hts[ndxblw]),
            float(parray[ndxblw]),
            ztop,
        )
        sum_val = sum_val + (ztop - float(hts[ndxblw - 1])) * 0.5 * (
            valtop + float(parray[ndxblw - 1])
        )
    else:
        sum_val = sum_val + (ztop - float(hts[nlvls - 1])) * float(parray[nlvls - 1])

    valavg = sum_val / (ztop - zbot)
    return valavg


def metini(state: AermodState) -> None:
    """SUBROUTINE METINI (iblval.f)."""
    from .metext import gintrp

    # Fortran NDXSTK(ISRC) -> state.NDXSTK[state.ISRC - 1]
    ndxstk = int(state.NDXSTK[state.ISRC - 1])

    if ndxstk >= 1:
        # Fortran NDXSTK(ISRC) -> idx = ndxstk - 1, NDXSTK(ISRC)+1 -> idx + 1
        idx = ndxstk - 1
        ht_lo = float(state.GRIDHT[idx])
        ht_hi = float(state.GRIDHT[idx + 1])

        # Sigma_V at stack height
        state.SVS = gintrp(
            ht_lo,
            float(state.GRIDSV[idx]),
            ht_hi,
            float(state.GRIDSV[idx + 1]),
            state.HS,
        )

        # Sigma_W
        state.SWS = gintrp(
            ht_lo,
            float(state.GRIDSW[idx]),
            ht_hi,
            float(state.GRIDSW[idx + 1]),
            state.HS,
        )

        # Wind speed
        state.US = gintrp(
            ht_lo,
            float(state.GRIDWS[idx]),
            ht_hi,
            float(state.GRIDWS[idx + 1]),
            state.HS,
        )

        # Wind direction: check for 360 crossover and adjust if necessary
        valabv = float(state.GRIDWD[idx + 1])
        vbelow = float(state.GRIDWD[idx])

        if (valabv - vbelow) < -180.0:
            valabv = valabv + 360.0
        elif (valabv - vbelow) > 180.0:
            valabv = valabv - 360.0

        if vbelow == valabv:
            state.WDIR = vbelow
        else:
            state.WDIR = gintrp(ht_lo, vbelow, ht_hi, valabv, state.HS)

        if state.WDIR > 360.0:
            state.WDIR = state.WDIR - 360.0
        elif state.WDIR <= 0.0:
            state.WDIR = state.WDIR + 360.0

        # Potential temperature gradient
        state.TGS = gintrp(
            ht_lo,
            float(state.GRIDTG[idx]),
            ht_hi,
            float(state.GRIDTG[idx + 1]),
            state.HS,
        )

        # Potential temperature
        state.PTS = gintrp(
            ht_lo,
            float(state.GRIDPT[idx]),
            ht_hi,
            float(state.GRIDPT[idx + 1]),
            state.HS,
        )
    else:
        # Use GRID value for lowest level
        state.SVS = float(state.GRIDSV[0])
        state.SWS = float(state.GRIDSW[0])
        state.US = float(state.GRIDWS[0])
        state.WDIR = float(state.GRIDWD[0])
        state.TGS = float(state.GRIDTG[0])
        state.PTS = float(state.GRIDPT[0])

    # Modify the treatment of low wind/low turbulence cases
    state.SWS = max(state.SWS, state.SWMIN)
    state.SVS = max(state.SVS, state.SVMIN, state.SVUMIN * state.US)
    if state.L_VECTORWS:
        state.US = math.sqrt(state.US * state.US + 2.0 * state.SVS * state.SVS)
    state.US = max(state.US, state.WSMIN)

    if not state.CLMHR and not state.MSGHR:
        state.WDSIN = math.sin(state.WDIR * state.DTORAD)
        state.WDCOS = math.cos(state.WDIR * state.DTORAD)

        state.AFV = state.WDIR - 180.0
        if state.AFV < 0.0:
            state.AFV = state.AFV + 360.0
        state.IFVSEC = f_int(state.AFV * 0.10 + 0.4999)
        if state.IFVSEC == 0:
            state.IFVSEC = 36

    # Assign TGP as TGS initially
    state.TGP = state.TGS

    # Calculate stack height ambient temperature, TA
    state.TA = state.PTS - state.GOVRCP * (state.HS + state.ZBASE)

    # Assign wind speed to use for plume rise, UP = US
    state.UP = state.US
    state.SVP = state.SVS
    state.SWP = state.SWS

    # Compute the Brunt-Vaisala frequency, BVF, at stack height
    if (state.TGS > 0.0) and (
        state.STABLE or (state.UNSTAB and state.HS >= state.ZI)
    ):
        state.BVF = math.sqrt(state.G * state.TGS / state.PTS)
    else:
        state.BVF = 1.0e-10

    if state.BVF < 1.0e-10:
        state.BVF = 1.0e-10

    state.BVPRIM = 0.7 * state.BVF

    # Set temporarily assigned effective values
    state.UEFF = state.US
    state.SVEFF = state.SVS
    state.SWEFF = state.SWS
    state.TGEFF = state.TGS
    state.UEFFD = state.US
    state.SVEFFD = state.SVS
    state.SWEFFD = state.SWS
    state.UEFFN = state.US
    state.SVEFFN = state.SVS
    state.SWEFFN = state.SWS
    state.UEFF3 = state.US
    state.SVEFF3 = state.SVS
    state.SWEFF3 = state.SWS
    state.TGEFF3 = state.TGS

    # Define temporary values of CENTER and SURFAC based on HS
    state.CENTER = state.HS
    if state.CENTER < 0.1 * state.ZI:
        state.SURFAC = True
    else:
        state.SURFAC = False

            ndxeff = locate(state.GRIDHT, 1, state.MXGLVL, state.HTEFF)
            # Fortran NDXEFF -> idx = ndxeff - 1, NDXEFF+1 -> idx + 1

def refsv(state: AermodState, htinp: float) -> float:
    """SUBROUTINE REFSV (siggrid.f)."""
    height = htinp
    value = 0.0

    if state.UNSTAB:
        svc = refsvc(state, height)
        svm = refsvm(state, height)
        value = math.sqrt(svc * svc + svm * svm)
    elif state.STABLE:
        svm = refsvm(state, height)
        value = svm

    return value


def refsvc(state: AermodState, htinp: float) -> float:
    """SUBROUTINE REFSVC (siggrid.f)."""
    from .metext import gintrp

    height = htinp
    zdcrs = state.AT1PT2 * state.ZICONV

    sv2 = 0.35 * state.WSTAR**2

    if height <= state.ZICONV:
        value = math.sqrt(sv2)
    elif state.ZICONV < height <= zdcrs:
        sv2dcr = min(sv2, 0.25)
        val2 = gintrp(state.ZICONV, sv2, zdcrs, sv2dcr, height)
        value = math.sqrt(val2)
    else:
        atzi = math.sqrt(sv2)
        value = min(atzi, 0.5)

    return value


def refsvm(state: AermodState, htinp: float) -> float:
    """SUBROUTINE REFSVM (siggrid.f)."""
    from .metext import gintrp

    height = htinp

    sv02 = 3.6 * state.USTAR**2
    sv2zi = min(sv02, 0.25)

    if height <= state.ZIMECH:
        varv = gintrp(0.0, sv02, state.ZIMECH, sv2zi, height)
        value = math.sqrt(varv)
    else:
        value = math.sqrt(sv2zi)

    return value


def ntrpsv(
    state: AermodState,

    htabov: float,
    vabove: float,
    reqdht: float,

def refswc(state: AermodState, htinp: float) -> float:
    """SUBROUTINE REFSWC (siggrid.f)."""
    height = htinp
    value = 0.0

    if height <= 0.1 * state.ZICONV:
        sw2 = (
            1.6
            * (height / state.ZICONV) ** (2.0 * state.THIRD)
            * state.WSTAR**2
        )
        value = math.sqrt(sw2)
    elif 0.1 * state.ZICONV < height <= state.ZICONV:
        value = math.sqrt(0.35 * state.WSTAR**2)
    elif height > state.ZICONV:
        exparg = -6.0 * (height - state.ZICONV) / state.ZICONV
        if exparg > state.EXPLIM:
            sw2 = 0.35 * state.WSTAR**2 * math.exp(exparg)
            value = math.sqrt(sw2)
        else:

    return value

"""SUBROUTINE COMPTG, TGINIT, GRDPTG, REFPTG, NTRPTG, XTRPTG, XTRPDN, GRDPT, GRDDEN (tempgrid.f)."""

def comptg(state: AermodState) -> None:

    state.PATH = "MX"

    ndxblw = 1
    ndxabv = 2
    state.NTGLVL = 0

    while ndxabv <= state.NPLVLS:
        # Fortran NDXBLW, NDXABV -> ndxblw - 1, ndxabv - 1
        b_idx = ndxblw - 1
        a_idx = ndxabv - 1
        if state.PFLTA[b_idx] > 0.0:
            if state.PFLTA[a_idx] > 0.0:
                state.NTGLVL = state.NTGLVL + 1
                # Fortran NTGLVL -> state.NTGLVL - 1
                tg_idx = state.NTGLVL - 1
                state.PFLTG[tg_idx] = (
                    float(state.PFLTA[a_idx]) - float(state.PFLTA[b_idx])
                ) / (
                    float(state.PFLHT[a_idx]) - float(state.PFLHT[b_idx])
                ) + state.GOVRCP
                state.PFLTGZ[tg_idx] = (
                    float(state.PFLHT[a_idx]) + float(state.PFLHT[b_idx])
                ) / 2.0
                ndxblw = ndxabv
                ndxabv = ndxabv + 1
            else:
                ndxabv = ndxabv + 1
        else:
            ndxabv = ndxabv + 1
            ndxblw = ndxblw + 1

    if state.STABLE:
        # Fortran NLVL = 1..NTGLVL -> nlvl in range(state.NTGLVL)
        for nlvl in range(state.NTGLVL):
            state.PFLTG[nlvl] = max(state.SPTGMN, float(state.PFLTG[nlvl]))

    if state.NTGLVL == 0:
        state.NTGLVL = 1


def tginit(state: AermodState) -> None:

def refptg(state: AermodState, htinp: float) -> float:
    """SUBROUTINE REFPTG (tempgrid.f)."""
    hdelth = 100.0
    height = htinp

    if state.UNSTAB:
        if height <= state.ZI:
            value = state.XVAL
        elif height <= state.ZI + 500.0:
            value = state.VPTGZI
        else:
            value = 0.005
    else:
        if height <= 2.0:
            value = state.TG4PFL
        elif height <= 100.0:
            if state.L_AdjUstar:
                value = (state.THSTAR / (state.VONKAR * height)) * (
                    0.74 + 4.7 * height / state.OBULEN
                )
            else:
                value = (state.THSTAR / (state.VONKAR * height)) * (
                    1.00 + 5.0 * height / state.OBULEN
                )
        else:
            exparg = -1.0 * (height - 100.0) / (
                state.EFOLDH * max(hdelth, state.ZI)
            )
            if exparg > state.EXPLIM:
                value = state.TG4XTR * math.exp(exparg)
            else:
                value = 0.0

        value = max(value, state.SPTGMN)

    return value

def xtrptg(state: AermodState, pflz: float, pflval: float, grdz: float) -> float:

def unstu(
    ustr: float, z: float, oblen: float, z0: float, valpi: float, valk: float
) -> float:
    """DOUBLE PRECISION FUNCTION UNSTU (windgrid.f)."""
    x0 = (1.0 - 16.0 * z / oblen) ** 0.25
    xz0 = (1.0 - 16.0 * z0 / oblen) ** 0.25

    psim = (
        2.0 * math.log((1.0 + x0) / 2.0)
        + math.log((1.0 + x0 * x0) / 2.0)
        - 2.0 * math.atan(x0)
        + valpi / 2.0
    )
    psimz0 = (
        2.0 * math.log((1.0 + xz0) / 2.0)
        + math.log((1.0 + xz0 * xz0) / 2.0)
        - 2.0 * math.atan(xz0)
        + valpi / 2.0
    )

    return (ustr / valk) * (math.log(z / z0) - psim + psimz0)


def stblu(ustr: float, z: float, oblen: float, z0: float, valk: float) -> float:
    """DOUBLE PRECISION FUNCTION STBLU (windgrid.f)."""
    psim = -17.0 * (1.0 - math.exp(-0.29 * z / oblen))
    psimz0 = -17.0 * (1.0 - math.exp(-0.29 * z0 / oblen))
    return (ustr / valk) * (math.log(z / z0) - psim + psimz0)

                state.GRIDTG[:] = state.GRDTGU[:, iurb_idx]
                state.GRIDPT[:] = state.GRDPTU[:, iurb_idx]
                state.OBULEN = float(state.URBOBULEN[iurb_idx])
                state.USTAR = float(state.URBUSTR[iurb_idx])
            else:
                state.URBSTAB = False
                state.ZI = state.ZIRUR
                state.GRIDSV[:] = state.GRDSVR[:]
                state.GRIDSW[:] = state.GRDSWR[:]
                state.GRIDTG[:] = state.GRDTGR[:]
                state.GRIDPT[:] = state.GRDPTR[:]
                state.OBULEN = state.RUROBULEN
                state.USTAR = state.RURUSTR
        elif state.URBAN and urbsrc_flag == "N":
            state.URBSTAB = False
            state.ZI = state.ZIRUR
            state.GRIDSV[:] = state.GRDSVR[:]
            state.GRIDSW[:] = state.GRDSWR[:]
            state.GRIDTG[:] = state.GRDTGR[:]
            state.GRIDPT[:] = state.GRDPTR[:]
            state.OBULEN = state.RUROBULEN
            state.USTAR = state.RURUSTR
        else:
            state.URBSTAB = False

        # Calculate the initial meteorological variables --- CALL METINI
        metini(state)

        # Calculate Buoyancy and Momentum Fluxes --- CALL FLUXES
        vseq = fluxes(state)

        # Set Wake and Building Type Switches --- CALL WAKFLG
        if state.OSPLAT[isrc_idx]:
            state.WAKE = False
        else:
            wakflg(state)

        # Define temporary values of CENTER and SURFAC based on HS
        state.CENTER = state.HS
        if state.CENTER < 0.1 * state.ZI:
            state.SURFAC = True
        else:
            state.SURFAC = False

        # Check for stack-tip downwash option and adjust if necessary
        srctyp = str(state.SRCTYP[isrc_idx]).strip()
        if srctyp == "POINTCAP":
            state.HSP = hsprim(state.US, vseq, state.HS, state.DS)
        elif srctyp == "POINTHOR":
            state.HSP = state.HS
        elif state.NOSTD:
            state.HSP = state.HS
        else:
            state.HSP = hsprim(state.US, state.VS, state.HS, state.DS)

        # Calculate Distance to Final Rise --- CALL DISTF
        distf(state)

        # Calculate the plume penetration factor --- CALL PENFCT
        penfct(state)

        if state.DEBUG:
            # WRITE(DBGUNT,6000) DHFAER, UP, TGS
            # Note: in MODEL.DBG, 6000 and 6001 are printed before 6010 (which resets
            # parser receptor/comp state after a previous source's receptor loop).
            if state.prev_active_isrc == 0:
                state.emit_debug(state.ISRC, 0, "", "PCALC", "DHFAER", state.DHFAER)
                state.emit_debug(state.ISRC, 0, "", "PCALC", "UP", state.UP)
                state.emit_debug(state.ISRC, 0, "", "PCALC", "TGS", state.TGS)
            else:
                state.emit_debug(
                    state.prev_active_isrc,
                    state.NUMREC,
                    "PAN",
                    "PCALC",
                    "DHFAER",
                    state.DHFAER,
                )
                state.emit_debug(
                    state.prev_active_isrc,
                    state.NUMREC,
                    "PAN",
                    "PCALC",
                    "UP",
                    state.UP,
                )
                state.emit_debug(
                    state.prev_active_isrc,
                    state.NUMREC,
                    "PAN",
                    "PCALC",
                    "TGS",
                    state.TGS,
                )

        if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
            # Use iterative approach to stable plume rise calculations
            kiter = 0
            while True:
                zplm = state.HSP + 0.5 * state.DHFAER
                dhfold = state.DHFAER

                # Locate index below ZPLM
                ndxzpl = locate(state.GRIDHT, 1, state.MXGLVL, zplm)
                # Fortran NDXZPL -> idx = ndxzpl - 1, NDXZPL+1 -> idx + 1
                idx = ndxzpl - 1
                ht_lo = float(state.GRIDHT[idx])
                ht_hi = float(state.GRIDHT[idx + 1])

                svpm = gintrp(
                    ht_lo,
                    float(state.GRIDSV[idx]),
                    ht_hi,
                    float(state.GRIDSV[idx + 1]),
                    zplm,
                )
                upm = gintrp(
                    ht_lo,
                    float(state.GRIDWS[idx]),
                    ht_hi,
                    float(state.GRIDWS[idx + 1]),
                    zplm,
                )
                svpm = max(svpm, state.SVMIN, state.SVUMIN * upm)
                if state.L_VECTORWS:
                    upm = math.sqrt(upm * upm + 2.0 * svpm * svpm)
                upm = max(upm, state.WSMIN)

                # Use average of stack top and midpoint wind speeds
                state.UP = 0.5 * (state.US + upm)

                tgpm = gintrp(
                    ht_lo,
                    float(state.GRIDTG[idx]),
                    ht_hi,
                    float(state.GRIDTG[idx + 1]),
                    zplm,
                )
                ptpm = gintrp(
                    ht_lo,
                    float(state.GRIDPT[idx]),
                    ht_hi,
                    float(state.GRIDPT[idx + 1]),
                    zplm,
                )
                # Use average of stack top and midpoint temperature gradients
                state.TGP = 0.5 * (state.TGS + tgpm)
                ptp = 0.5 * (state.PTS + ptpm)
                state.BVF = math.sqrt(state.G * state.TGP / ptp)
                if state.BVF < 1.0e-10:
                    state.BVF = 1.0e-10
                state.BVPRIM = 0.7 * state.BVF

                distf(state)

                kiter = kiter + 1

                if state.DEBUG:
                    # WRITE(DBGUNT,6001) KITER, DHFOLD, DHFAER, ZPLM, UP, TGP
                    if state.prev_active_isrc == 0:
                        state.emit_debug(state.ISRC, 0, "", "PCALC", "DHFOLD", dhfold)
                        state.emit_debug(
                            state.ISRC, 0, "", "PCALC", "DHFAER", state.DHFAER
                        )
                        state.emit_debug(state.ISRC, 0, "", "PCALC", "ZPLM", zplm)
                        state.emit_debug(state.ISRC, 0, "", "PCALC", "UP", state.UP)
                        state.emit_debug(state.ISRC, 0, "", "PCALC", "TGP", state.TGP)
                    else:
                        state.emit_debug(
                            state.prev_active_isrc,
                            state.NUMREC,
                            "PAN",
                            "DELTAH",
                            "DHPOLD",
                            dhfold,
                        )
                        state.emit_debug(
                            state.prev_active_isrc,
                            state.NUMREC,
                            "PAN",
                            "DELTAH",
                            "DHP",
                            state.DHFAER,
                        )
                        state.emit_debug(
                            state.prev_active_isrc,
                            state.NUMREC,
                            "PAN",
                            "DELTAH",
                            "ZPLM",
                            zplm,
                        )
                        state.emit_debug(
                            state.prev_active_isrc,
                            state.NUMREC,
                            "PAN",
                            "DELTAH",
                            "UP",
                            state.UP,
                        )
                        state.emit_debug(
                            state.prev_active_isrc,
                            state.NUMREC,
                            "PAN",
                            "DELTAH",
                            "TGP",
                            state.TGP,
                        )

                # Check for convergence
                if abs((dhfold - state.DHFAER) / state.DHFAER) < 0.01:
                    break

                if kiter >= 5:
                    state.DHFAER = 0.5 * (state.DHFAER + dhfold)
                    break

            # After completing iteration, reset UP and TGP to stack top values
            state.UP = state.US
            state.TGP = state.TGS
            ptp = state.PTS
            state.BVF = math.sqrt(state.G * state.TGP / ptp)
            if state.BVF < 1.0e-10:
                state.BVF = 1.0e-10
            state.BVPRIM = 0.7 * state.BVF

        state.PRM_FSTREC = True
        state.ZMIDMX = 0.5 * state.ZI

        if state.DEBUG:
            # WRITE(DBGUNT,6010) KURDAT, ZMIDMX
            state.emit_debug(state.ISRC, 0, "", "PCALC", "ZMIDMX", state.ZMIDMX)

        state.XFINAL = state.XMAX
        state.DHCRIT = state.DHFAER
        state.XMIXED = state.ZI * state.UAVG / state.SWAVG
        if state.UNSTAB and state.HS < state.ZI:
            if state.XMIXED < 1.25 * state.XFINAL:
                state.XFINAL = 0.8 * state.XMIXED
                cblprd(state, state.XFINAL)
                state.DHCRIT = state.DHP1

        # Initialize PDF parameters for use in calculating ZSUBP
        if state.UNSTAB and (state.HS < state.ZI):
            pdf(state)

        # Determine transport wind direction using midpoint between stack height and "final" plume height
        zmid = min(4000.0, (state.HS + 0.5 * state.DHFAER))

        ndxzmid = locate(state.GRIDHT, 1, state.MXGLVL, zmid)
        # Fortran NDXZMID -> idx = ndxzmid - 1, NDXZMID+1 -> idx + 1
        idx_zmid = ndxzmid - 1
        valabv = float(state.GRIDWD[idx_zmid + 1])
        vbelow = float(state.GRIDWD[idx_zmid])

        if (valabv - vbelow) < -180.0:
            valabv = valabv + 360.0
        elif (valabv - vbelow) > 180.0:
            valabv = valabv - 360.0

        if vbelow == valabv:
            state.WDIR = vbelow
        else:
            state.WDIR = gintrp(
                float(state.GRIDHT[idx_zmid]),
                vbelow,
                float(state.GRIDHT[idx_zmid + 1]),
                valabv,
                zmid,
            )

        if state.WDIR > 360.0:
            state.WDIR = state.WDIR - 360.0
        elif state.WDIR <= 0.0:
            state.WDIR = state.WDIR + 360.0

        state.WDSIN = math.sin(state.WDIR * state.DTORAD)
        state.WDCOS = math.cos(state.WDIR * state.DTORAD)

        state.AWDSIN[isrc_idx] = state.WDSIN
        state.AWDCOS[isrc_idx] = state.WDCOS

        state.AFV = state.WDIR - 180.0
        if state.AFV < 0.0:
            state.AFV = state.AFV + 360.0
        state.AAFV[isrc_idx] = state.AFV
        state.IFVSEC = f_int(state.AFV * 0.10 + 0.4999)
        if state.IFVSEC == 0:
            state.IFVSEC = 36

        if state.DEBUG:
            # WRITE(DBGUNT, 6011) DHCRIT, XFINAL, XMIXED, AFV
            state.emit_debug(state.ISRC, 0, "", "PCALC", "DHCRIT", state.DHCRIT)
            state.emit_debug(state.ISRC, 0, "", "PCALC", "XFINAL", state.XFINAL)
            state.emit_debug(state.ISRC, 0, "", "PCALC", "XMIXED", state.XMIXED)
            state.emit_debug(state.ISRC, 0, "", "PCALC", "AFV", state.AFV)
            psrdeb(state)

        # Begin Receptor LOOP
        # Fortran IREC = 1..NUMREC -> irec in range(state.NUMREC)
        for irec in range(state.NUMREC):
            state.IREC = irec + 1

            if state.EVONLY:
                xydist(state, state.IEVENT)
            else:
                xydist(state, state.IREC)

            if state.DISTR < 0.99 or state.DISTR > state.MAXDIST:
                state.HRVAL[:] = 0.0
                aerplm[:] = 0.0
                aerpan[:] = 0.0
                state.PRMVAL[:] = 0.0
                continue

            # Calculate coherent plume component using downwind distance
            l_plume = True
            state.XDIST = state.X
            aercalc(state, state.X, l_plume, aerplm)

            if state.L_EFFSIGY:
                state.AERVAL[:] = aerplm[:]
                aerpan[:] = 0.0
                fran = 0.0
            else:
                # Next calculate random "pancake" component using radial distance
                l_plume = False
                state.XDIST = state.DISTR
                aercalc(state, state.DISTR, l_plume, aerpan)

                state.debug_comp = "PAN"
                if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
                    fran = meandr(state, state.UEFF, state.SVEFF)
                elif state.UNSTAB:
                    fran = meandr(state, state.UEFFD, state.SVEFFD)
                    if state.PPF > 0.0:
                        fran3 = meandr(state, state.UEFF3, state.SVEFF3)
                        fran = state.PPF * fran3 + (1.0 - state.PPF) * fran

                state.AERVAL[:] = fran * aerpan + (1.0 - fran) * aerplm

            if state.DEBUG:
                # Fortran DO ITYP = 1, NUMTYP -> ityp in range(state.NUMTYP)
                for ityp in range(state.NUMTYP):
                    # WRITE(DBGUNT,10) AERPAN(ITYP), AERPLM(ITYP), FRAN, AERVAL(ITYP)
                    state.emit_debug(
                        state.ISRC,
                        state.IREC,
                        "PAN",
                        "PCALC",
                        "AERPAN",
                        float(aerpan[ityp]),
                    )
                    state.emit_debug(
                        state.ISRC,
                        state.IREC,
                        "PAN",
                        "PCALC",
                        "AERPLM",
                        float(aerplm[ityp]),
                    )
                    state.emit_debug(
                        state.ISRC, state.IREC, "PAN", "PCALC", "FRAN", fran
                    )
                    state.emit_debug(
                        state.ISRC,
                        state.IREC,
                        "PAN",
                        "PCALC",
                        "AERVAL",
                        float(state.AERVAL[ityp]),
                    )

            # Tier A has no building downwash (WAKE is False)
            state.GAMFACT = 0.0
            state.HRVAL[:] = state.AERVAL[:]
            state.PRMVAL[:] = 0.0

            # Sum HRVAL to AVEVAL and ANNVAL Arrays --- CALL SUMVAL
            if not state.EVONLY:
                # Fortran IGRP = 1..NUMGRP -> igrp in range(state.NUMGRP)
                for igrp in range(state.NUMGRP):
                    state.IGRP = igrp + 1
                    sumval(state)

            # Initialize __VAL arrays (1:NUMTYP)
            state.HRVAL[:] = 0.0
            state.AERVAL[:] = 0.0
            aerplm[:] = 0.0
            aerpan[:] = 0.0
            state.PRMVAL[:] = 0.0

        state.prev_active_isrc = state.ISRC


def aercalc(
    state: AermodState, xarg: float, l_plume: bool, aerout: xp.ndarray
) -> None:
    """SUBROUTINE AERCALC (calc1.f)."""
    aerout[:] = 0.0
    aertmp = xp.zeros(state.NUMTYP, dtype=xp.float64)

    fran = 0.0
    fran3 = 0.0

    if xarg < 1.0:
        aerout[:] = 0.0
        return

    state.debug_comp = "PLM" if l_plume else "PAN"
    comp = state.debug_comp

    # Determine Deposition Correction Factors
    if state.NPD == 0 and (state.LDGAS or state.LWGAS):
        raise RuntimeError("Gas deposition out of scope")
    else:
        state.DQCORG = 1.0
        state.WQCORG = 1.0

    # Set initial effective parameters
    state.UEFF = state.US
    state.SVEFF = state.SVS
    state.SWEFF = state.SWS
    state.TGEFF = state.TGS
    if state.UNSTAB and (state.HS < state.ZI):
        state.UEFFD = state.US
        state.SVEFFD = state.SVS
        state.SWEFFD = state.SWS
        state.UEFFN = state.US
        state.SVEFFN = state.SVS
        state.SWEFFN = state.SWS
        state.UEFF3 = state.US
        state.SVEFF3 = state.SVS
        state.SWEFF3 = state.SWS
        state.TGEFF3 = state.TGS

    if state.DEBUG:
        # WRITE(DBGUNT, 6015 / 6016) UEFF, SVEFF, SWEFF
        # Parsed by dbgparse.py as IBLVAL UEFF, SVEFF, SWEFF
        state.emit_debug(state.ISRC, state.IREC, comp, "IBLVAL", "UEFF", state.UEFF)
        state.emit_debug(state.ISRC, state.IREC, comp, "IBLVAL", "SVEFF", state.SVEFF)
        state.emit_debug(state.ISRC, state.IREC, comp, "IBLVAL", "SWEFF", state.SWEFF)

    # Define plume centroid height (CENTER) for use in inhomogeneity calculations
    centroid(state, xarg)

    # Calculate the plume rise --- CALL DELTAH
    deltah(state, xarg)

    # If the atmosphere is unstable and the stack top is below the mixing height,
    # calculate the CBL PDF coefficients --- CALL PDF
    if state.UNSTAB and (state.HS < state.ZI):
        pdf(state)

    # Determine Effective Plume Height --- CALL HEFF
    heff(state, xarg)

    # Compute effective parameters using an average through plume layer
    iblval(state, xarg)

    # Call PDF & HEFF again for final CBL plume heights
    if state.UNSTAB and (state.HS < state.ZI):
        pdf(state)
        heff(state, xarg)

    # Determine Dispersion Parameters --- CALL PDIS
    pdis(state, xarg)

    # Calculate the 'y-term' contribution to dispersion, FSUBY
    if l_plume:
        if state.L_EFFSIGY:
            if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
                fran = meandr(state, state.UEFF, state.SVEFF)
            elif state.UNSTAB:
                fran = meandr(state, state.UEFFD, state.SVEFFD)
                if state.PPF > 0.0:
                    fran3 = meandr(state, state.UEFF3, state.SVEFF3)
                    fran = state.PPF * fran3 + (1.0 - state.PPF) * fran

            state.SYEFF = 1.0 / (
                (fran / (state.SRT2PI * xarg)) + (1.0 - fran) / state.SY
            )
            if abs(state.Y) > state.NUMSYEFF * state.SYEFF:
                fyout = 0.0
            else:
                fyout = fyplm(state, state.SYEFF)
        else:
            fyout = fyplm(state, state.SY)
    else:
        fyout = fypan(state)

    state.FSUBY = fyout
    state.FSUBYD = state.FSUBY
    state.FSUBYN = state.FSUBYD

    # Calculate the 'y-term' contribution to dispersion for the penetrated plume, FSUBY3
    if state.UNSTAB and (state.HS < state.ZI) and (state.PPF > 0.0):
        if l_plume:
            if state.L_EFFSIGY:
                state.SYEFF = 1.0 / (
                    (fran / (state.SRT2PI * xarg)) + (1.0 - fran) / state.SY3
                )
                if abs(state.Y) > state.NUMSYEFF * state.SYEFF:
                    fyout = 0.0
                else:
                    fyout = fyplm(state, state.SYEFF)
            else:
                fyout = fyplm(state, state.SY3)
        else:
            fyout = fypan(state)
        state.FSUBY3 = fyout
    else:
        state.FSUBY3 = 0.0

    # Check for zero "y-terms"; if zero then skip calculations
    if state.FSUBY == 0.0 and state.FSUBY3 == 0.0:
        aerout[:] = 0.0
    else:
        if state.NPD == 0:
            # Perform calculations for gases
            state.HV = 0.0
            adj = state.DQCORG * state.WQCORG

            if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
                state.HSBL = refl_ht(state, state.HE, xarg, state.SZB, 0.0)
            elif state.UNSTAB:
                state.HSBL = 0.0

            if state.UNSTAB and (state.HS < state.ZI) and (state.PPF > 0.0):
                state.HPEN = refl_ht(state, state.HE3, xarg, state.SZB3, 0.0)
            else:
                state.HPEN = 0.0

            # Determine the CRITical Dividing Streamline --- CALL CRITDS
            critds(state, state.HE)

            # Calculate the fraction of plume below HCRIT, PHEE --- CALL PFRACT
            pfract(state, state.HE)

            # Calculate FOPT = f(PHEE) --- CALL FTERM
            fterm(state)

            # Calculate AERMOD Concentration --- CALL AER_PCHI
            aer_pchi(state, xarg, adj, state.VDEPG, 0, aerout)
        else:
            # Perform calculations for particles
            # Fortran J = 1..NPD -> j in range(state.NPD)
            for j in range(state.NPD):
                state.HV = (xarg / state.US) * float(state.VGRAV[j])
                adj = (
                    float(state.PHI[j])
                    * float(state.DQCOR[j])
                    * float(state.WQCOR[j])
                )
                if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
                    state.HESETL = max(0.0, state.HE - state.HV)
                    state.HSBL = refl_ht(state, state.HESETL, xarg, state.SZB, 0.0)
                elif state.UNSTAB:
                    state.HESETL = max(
                        0.0, 0.5 * (state.HED1 + state.HED2) - state.HV
                    )
                    state.HSBL = 0.0

                if state.UNSTAB and (state.HS < state.ZI) and (state.PPF > 0.0):
                    state.HE3SETL = max(0.0, state.HE3 - state.HV)
                    state.HPEN = refl_ht(state, state.HE3SETL, xarg, state.SZB3, 0.0)
                    state.HPEN = max(state.HPEN, state.ZI)
                else:
                    state.HPEN = 0.0

                critds(state, state.HESETL)
                pfract(state, state.HESETL)
                fterm(state)
                aer_pchi(state, xarg, adj, float(state.VDEP[j]), j + 1, aertmp)
                aerout[:] = aerout + aertmp


def centroid(state: AermodState, xarg: float) -> None:
    """SUBROUTINE CENTROID (calc1.f)."""
    xtmp = max(xarg, 1.0)
    # Fortran ISRC (1-based) -> state.ISRC - 1
    srctyp = str(state.SRCTYP[state.ISRC - 1]).strip()

    if state.UNSTAB and (state.HS < state.ZI):
        if srctyp[:5] == "POINT":
            cblprd(state, xtmp)
            state.HTEFF = min(state.HSP + state.DHP1, state.ZI)
        else:
            state.HTEFF = state.HSP
            state.DHCRIT = 0.0
            state.XFINAL = 0.0

        if xtmp < state.XFINAL:
            state.CENTER = state.HTEFF
        elif xtmp >= state.XMIXED:
            state.CENTER = state.ZMIDMX
        else:
            delx = state.XMIXED - state.XFINAL
            delz = state.ZMIDMX - min(state.HSP + state.DHCRIT, state.ZI)
            frac = (xtmp - state.XFINAL) / delx
            state.CENTER = min(state.HSP + state.DHCRIT, state.ZI) + frac * delz

        if state.CENTER < 0.1 * state.ZI:
            state.SURFAC = True
        else:
            state.SURFAC = False
    elif state.UNSTAB and (state.HS >= state.ZI):
        state.SURFAC = False
        state.CENTER = state.HSP
        if state.CENTER < 0.1 * state.ZI:
            state.SURFAC = True
        else:
            state.SURFAC = False
    else:
        state.CENTER = state.HSP
        if state.CENTER < 0.1 * state.ZI:
            state.SURFAC = True
        else:
            state.SURFAC = False


def refl_ht(
    state: AermodState,
    hearg: float,
    xarg: float,
    szbarg: float,
    vsigzarg: float,
) -> float:
    """SUBROUTINE REFL_HT (calc1.f)."""
    ndxhe = locate(state.GRIDHT, 1, state.MXGLVL, hearg)

    if ndxhe >= 1:
        # Fortran NDXHE -> idx = ndxhe - 1, NDXHE+1 -> idx + 1
        idx = ndxhe - 1
        ht_lo = float(state.GRIDHT[idx])
        ht_hi = float(state.GRIDHT[idx + 1])

        svrefl = gintrp(
            ht_lo,
            float(state.GRIDSV[idx]),
            ht_hi,
            float(state.GRIDSV[idx + 1]),
            hearg,
        )
        swrefl = gintrp(
            ht_lo,
            float(state.GRIDSW[idx]),
            ht_hi,
            float(state.GRIDSW[idx + 1]),
            hearg,
        )
        urefl = gintrp(
            ht_lo,
            float(state.GRIDWS[idx]),
            ht_hi,
            float(state.GRIDWS[idx + 1]),
            hearg,
        )
        tgrefl = gintrp(
            ht_lo,
            float(state.GRIDTG[idx]),
            ht_hi,
            float(state.GRIDTG[idx + 1]),
            hearg,
        )
        ptrefl = gintrp(
            ht_lo,
            float(state.GRIDPT[idx]),
            ht_hi,
            float(state.GRIDPT[idx + 1]),
            hearg,
        )
    else:
        svrefl = float(state.GRIDSV[0])
        swrefl = float(state.GRIDSW[0])
        urefl = float(state.GRIDWS[0])
        tgrefl = float(state.GRIDTG[0])
        ptrefl = float(state.GRIDPT[0])

    swrefl = max(swrefl, state.SWMIN)
    svrefl = max(svrefl, state.SVMIN, state.SVUMIN * urefl)
    if state.L_VECTORWS:
        urefl = math.sqrt(urefl * urefl + 2.0 * svrefl * svrefl)
    urefl = max(urefl, state.WSMIN)

    if state.STABLE:
        state.SZSURF = (
            (state.RTOF2 / state.RTOFPI)
            * state.USTAR
            * (xarg / urefl)
            * (1.0 + 0.7 * xarg / state.OBULEN) ** (-1.0 * state.THIRD)
        )
    else:
        state.SZSURF = 0.0

    ttravl = xarg / urefl
    bvfrq = math.sqrt(state.G * tgrefl / ptrefl)
    if bvfrq < 1.0e-10:
        bvfrq = 1.0e-10

    ztmp = max(state.HS, hearg, 1.0e-4)
    szrefl = (
        swrefl
        * ttravl
        / math.sqrt(
            1.0
            + swrefl
            * ttravl
            * (1.0 / (0.72 * ztmp) + bvfrq / (0.54 * swrefl))
        )
    )

    if hearg >= state.ZI:
        szrf = szrefl
    else:
        sigf = min(hearg / state.ZI, 1.0)
        szrf = (1.0 - sigf) * state.SZSURF + sigf * szrefl

    szhsbl = math.sqrt(szbarg * szbarg + szrf * szrf + vsigzarg * vsigzarg)
    herefl = max(state.ZI, hearg + 2.15 * szhsbl)
    return herefl

def setsrc(state: AermodState) -> None:
    """SUBROUTINE SETSRC (calc1.f)."""
    state.XINIT = 0.0
    state.YINIT = 0.0
    state.SYINIT = 0.0
    state.SZINIT = 0.0
    state.ANGLE = 0.0
    state.PALPHA = 0.0
    state.PDEFF = 0.0
    state.PITLEN = 0.0
    state.PITWID = 0.0

    # Fortran ISRC (1-based) -> idx = state.ISRC - 1
    idx = state.ISRC - 1
    srctyp = str(state.SRCTYP[idx]).strip()

    if srctyp[:5] == "POINT":
        state.XS = float(state.AXS[idx])
        state.YS = float(state.AYS[idx])
        state.ZS = float(state.AZS[idx])
        state.QS = float(state.AQS[idx])
        state.HS = float(state.AHS[idx])

        state.DS = float(state.ADS[idx])
        state.VS = float(state.AVS[idx])
        state.TS = float(state.ATS[idx])

        if srctyp == "POINTCAP":
            state.DSFACT = float(state.ADSFACT[idx])
        else:
            state.DSFACT = 1.0

    state.NPD = int(state.INPD[idx])
    if state.NPD > 0:
        # Fortran J = 1..NPD -> j in range(state.NPD)
        state.PDIAM = state.APDIAM[: state.NPD, idx].copy()
        state.PHI = state.APHI[: state.NPD, idx].copy()

        state.VGRAV = state.AVGRAV[: state.NPD, idx].copy()
        state.TSTOP = state.ATSTOP[: state.NPD, idx].copy()

    if state.HS < 0.1 * state.ZI:
        state.SURFAC = True
    else:
        state.SURFAC = False


def fluxes(state: AermodState) -> float:
    """SUBROUTINE FLUXES (calc1.f)."""
    # Check for Negative Stack Temperature, Used to Indicate Constant TS-TA
    if state.TS < 0.0:
        state.TS = state.TA + abs(state.TS)

    if state.TS < state.TA:
        state.TS = state.TA
    state.FB = (
        (0.25 / state.TS)
        * (state.VS * state.DS * state.DS)
        * state.G
        * (state.TS - state.TA)
    )
    state.FM = (
        (0.25 / state.TS)
        * (state.VS * state.DS * state.DS)
        * state.VS
        * state.TA
    )

    # Fortran ISRC (1-based) -> state.ISRC - 1
    srctyp = str(state.SRCTYP[state.ISRC - 1]).strip()
    if srctyp == "POINTCAP":
        vseq = 0.001
        dseq = state.DS * math.sqrt(state.VS / vseq)
        tseq = state.TS
        state.FM = (0.25 / tseq) * (vseq * dseq * dseq) * vseq * state.TA
    elif srctyp == "POINTHOR":
        vseq = 0.001
        dseq = state.DS * math.sqrt(state.VS / vseq)
        tseq = state.TS
        state.FM = (0.25 / tseq) * (vseq * dseq * dseq) * vseq * state.TA
    else:
        vseq = state.VS

    if state.FB < 1.0e-10:
        state.FB = 1.0e-10
    if state.FM < 1.0e-10:
        state.FM = 1.0e-10

    return vseq


def heff(state: AermodState, xarg: float) -> None:
    """SUBROUTINE HEFF (calc1.f)."""
    state.DHP3PLAT = 0.0
    # Fortran ISRC (1-based) -> isrc_idx = state.ISRC - 1
    isrc_idx = state.ISRC - 1

    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        state.HE = state.HSP + state.DHP
        state.HE = max(0.0, state.HE)

        state.HED1 = 0.0
        state.HED2 = 0.0
        state.HEN1 = 0.0
        state.HEN2 = 0.0
        state.HE3 = 0.0
    elif state.UNSTAB:
        state.HE = 0.0

        state.HED1 = state.HSP + state.DHP1 + (state.ASUB1 * state.WSTAR * xarg / state.UEFFD)
        state.HED2 = state.HSP + state.DHP1 + (state.ASUB2 * state.WSTAR * xarg / state.UEFFD)

        state.HEN1 = (
            state.HSP
            + state.DHP1
            - state.DHP2
            + (state.ASUB1 * state.WSTAR * xarg / state.UEFFN)
        )
        state.HEN2 = (
            state.HSP
            + state.DHP1
            - state.DHP2
            + (state.ASUB2 * state.WSTAR * xarg / state.UEFFN)
        )

        if state.PPF > 0.0:
            if state.PPF == 1.0:
                state.DHP3 = state.HEDHH * (state.ZI - state.HSP)
            else:
                state.DHP3 = (
                    0.75 * (state.ZI - state.HSP) * state.HEDHH
                    + 0.5 * (state.ZI - state.HSP)
                )

            if state.OSPLAT[isrc_idx]:
                state.HE3 = state.HSP + state.DHP3 - state.DHP3PLAT
            else:
                state.HE3 = state.HSP + state.DHP3
        else:
            state.DHP3 = 0.0
            state.HE3 = 0.0


def pdis(state: AermodState, xarg: float) -> None:
    """SUBROUTINE PDIS (calc1.f)."""
    # Calculate Sigma-z from formulae --- CALL SIGZ
    sigz(state, xarg)
    # Calculate Sigma-y from formulae --- CALL SIGY
    sigy(state, xarg)

    # Set all virtual source terms to 0 for non-downwashing sources
    state.VSIGY = 0.0
    state.VSYN = 0.0
    state.VSIGZ = 0.0
    state.VSZD1 = 0.0
    state.VSZD2 = 0.0
    state.VSZN1 = 0.0
    state.VSZN2 = 0.0
    state.VSZ3 = 0.0
    state.VSY3 = 0.0

    # Calculate the buoyancy-induced dispersion parameters
    if state.NOBID:
        if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
            state.SYB = 0.0
            state.SZB = 0.0
        elif state.UNSTAB:
            state.SYB = 0.0
            state.SZBD = 0.0
            state.SZBN = 0.0
            state.SYB3 = 0.0
            state.SZB3 = 0.0
    else:
        bid(state)

    # Initialize platform downwash sigmas
    state.PLATSZ = 0.0
    state.PLATSY = 0.0
    state.PLATSZD1 = 0.0
    state.PLATSYD1 = 0.0
    state.PLATSZD2 = 0.0
    state.PLATSYD2 = 0.0
    state.PLATSZN1 = 0.0
    state.PLATSYN1 = 0.0
    state.PLATSZN2 = 0.0
    state.PLATSYN2 = 0.0
    state.PLATSZP = 0.0
    state.PLATSYP = 0.0

    # Calculate the root-mean-square sigma_Y and sigma_Z --- CALL RMSSIG
    rmssig(state)


def vdis(state: AermodState, xarg: float) -> None:
    """SUBROUTINE VDIS (calc1.f)."""
    sigy(state, xarg)
    sigz(state, xarg)

    state.VSIGY = state.SYINIT
    state.VSYN = state.SYINIT
    state.VSIGZ = state.SZINIT
    state.VSZD1 = state.SZINIT
    state.VSZD2 = state.SZINIT
    state.VSZN1 = state.SZINIT
    state.VSZN2 = state.SZINIT
    state.VSZ3 = 0.0
    state.VSY3 = 0.0

    # Fortran ISRC -> state.ISRC - 1
    if str(state.AFTSRC[state.ISRC - 1]).strip() == "Y":
        bid(state)
    else:
        state.SYB = 0.0
        state.SZB = 0.0
        state.SZBD = 0.0
        state.SZBN = 0.0
        state.SYB3 = 0.0
        state.SZB3 = 0.0

    rmssig(state)

def aer_pchi(
    state: AermodState,
    xarg: float,
    adj: float,
    vdinp: float,
    jin: int,
    aerout: xp.ndarray,
) -> None:
    """SUBROUTINE AER_PCHI (calc1.f)."""
    # Calculate the contribution due to horizontal plume, CWRAP
    if state.FOPT == 0.0:
        state.CWRAP = 0.0
    else:
        state.CWRAP = cplume(state, state.ZRT)

    # Calculate the contribution due to terrain-following plume, CLIFT
    if state.ZRT == state.ZFLAG:
        state.CLIFT = state.CWRAP
    elif state.FOPT == 1.0:
        state.CLIFT = 0.0
    else:
        state.CLIFT = cplume(state, state.ZFLAG)

    # Calculate the exponential decay term, D --- CALL DECAY
    decay(state, xarg)

    state.ITYP = 0
    if state.CONC:
        state.ITYP = 1
        # Fortran ITYP = 1 -> index 0
        aerout[0] = (
            adj
            * float(state.EMIFAC[0])
            * (state.FOPT * state.CWRAP + (1.0 - state.FOPT) * state.CLIFT)
            * state.D
        )

        if state.DEBUG:
            # WRITE(DBGUNT,10) ITYP, ADJ, FOPT, CWRAP, CLIFT, D, AEROUT(ITYP)
            comp = state.debug_comp
            isrc = state.ISRC
            irec = state.IREC
            state.emit_debug(isrc, irec, comp, "AER_PCHI", "ADJ", adj)
            state.emit_debug(isrc, irec, comp, "AER_PCHI", "FOPT", state.FOPT)
            state.emit_debug(isrc, irec, comp, "AER_PCHI", "CWRAP", state.CWRAP)
            state.emit_debug(isrc, irec, comp, "AER_PCHI", "CLIFT", state.CLIFT)
            state.emit_debug(isrc, irec, comp, "AER_PCHI", "D", state.D)
            state.emit_debug(
                isrc, irec, comp, "AER_PCHI", "AEROUT", float(aerout[0])
            )

    if state.METEORDBG:
        metdeb(state)

    if state.DEBUG:
        debout(state)


def debout(state: AermodState) -> None:

    emifac1 = float(state.EMIFAC[0])
    chid = 0.0
    chin = 0.0
    chi3 = 0.0

    # Calculate contributions from each "plume"
    if state.STABLE or (state.UNSTAB and state.HS >= state.ZI):
        # Fortran HRVAL(1) -> state.HRVAL[0]
        chid = float(state.HRVAL[0])
    elif state.UNSTAB:
        chid = (
            (state.QTK * emifac1 * (1.0 - state.PPF) / state.UEFFD)
            * (state.FSUBYD * state.FSUBZD)
        )
        chin = (
            (state.QTK * emifac1 * (1.0 - state.PPF) / state.UEFFN)
            * (state.FSUBYN * state.FSUBZN)
        )
        if state.PPF > 0.0 and state.UEFF3 > 0.0:
            chi3 = (
                (state.QTK * emifac1 * state.PPF / state.UEFF3)
                * (state.FSUBY3 * state.FSUBZ3)
            )
        else:
            chi3 = 0.0

    isrc = state.ISRC
    irec = state.IREC
    comp = state.debug_comp

    # WRITE ( DBGUNT , 320 ) IREC, XR, YR, ZELEV, ZHILL, ZFLAG,
    # X, Y, ZELEV-ZS, HCRIT, PHEE, FOPT,

def penfct(state: AermodState) -> None:
    """SUBROUTINE PENFCT (calc1.f)."""
    if state.STABLE:
        state.PPF = 0.0
    elif state.UNSTAB and (state.HS >= state.ZI):
        state.PPF = 1.0
    else:
        bvzi2 = (state.G / state.PTATZI) * state.VPTGZI
        # Fortran ISRC -> state.ISRC - 1
        if str(state.AFTSRC[state.ISRC - 1]).strip() == "Y":
            state.PSUBS = state.FB / (
                bvzi2
                * (state.ZI - state.HSP)
                * (state.ZI - state.HSP)
                * (state.ZI - state.HSP)
            )
        else:
            state.PSUBS = state.FB / (
                state.UP
                * bvzi2
                * (state.ZI - state.HSP)
                * (state.ZI - state.HSP)
                * (state.ZI - state.HSP)
            )

        state.HEDHH = (17.576 * state.PSUBS + 0.296296) ** state.THIRD

        if state.HEDHH < (2.0 * state.THIRD):
            state.PPF = 0.0
        elif state.HEDHH > 2.0:
            state.PPF = 1.0
        else:
            state.PPF = 1.5 - (1.0 / state.HEDHH)


def cplume(state: AermodState, zarg: float) -> float:
    """SUBROUTINE CPLUME (calc1.f)."""
    state.ZR = zarg
    cout = 0.0

    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        if state.ZR <= state.HSBL:
            vrtsbl(state, state.SZ, max(0.0, state.HE - state.HV), state.HSBL)
        else:
            vrtsbn(state, state.SZ, max(0.0, state.HE - state.HV))

        cout = (state.QTK / state.UEFF) * (state.FSUBY * state.FSUBZ)

    elif state.UNSTAB:
        if state.PPF < 1.0:
            if state.ZR <= state.ZI:
                vrtcbl(
                    state,
                    state.HED1 - state.HV,
                    state.HED2 - state.HV,
                    state.SZD1,
                    state.SZD2,
                    1.0,
                )
                state.FSUBZD = state.FSUBZ
            else:
                state.FSUBZD = 0.0

            if state.ZR <= state.ZI:
                vrtcbl(
                    state,
                    state.HEN1 - state.HV,
                    state.HEN2 - state.HV,
                    state.SZN1,
                    state.SZN2,
                    -1.0,
                )
                state.FSUBZN = state.FSUBZ
            else:
                state.FSUBZN = 0.0
        else:
            state.FSUBZD = 0.0
            state.FSUBZN = 0.0

        if state.PPF > 0.0:
            if state.ZR <= state.HPEN:
                vrtsbl(state, state.SZ3, max(0.0, state.HE3 - state.HV), state.HPEN)
            else:
                vrtsbn(state, state.SZ3, max(0.0, state.HE3 - state.HV))
            state.FSUBZ3 = state.FSUBZ

            if state.PPF < 1.0:
                cout = (
                    (state.QTK * (1.0 - state.PPF) / state.UEFFD)
                    * (state.FSUBYD * state.FSUBZD)
                    + (state.QTK * (1.0 - state.PPF) / state.UEFFN)
                    * (state.FSUBYN * state.FSUBZN)
                    + (state.QTK * state.PPF / state.UEFF3)
                    * (state.FSUBY3 * state.FSUBZ3)
                )
            else:
                cout = (state.QTK * state.PPF / state.UEFF3) * (
                    state.FSUBY3 * state.FSUBZ3
                )
        else:
            state.FSUBZ3 = 0.0
            state.HPEN = 0.0
            cout = (state.QTK / state.UEFFD) * (state.FSUBYD * state.FSUBZD) + (
                state.QTK / state.UEFFN
            ) * (state.FSUBYN * state.FSUBZN)

    return cout


def ltopg(state: AermodState) -> int:
    """SUBROUTINE LTOPG (calc1.f)."""

    def xl(yy: float, xm: float, b: float) -> float:
        return xm / (math.log(yy) - b)

    el = state.OBULEN
    zr0 = state.SFCZ0

    z0 = zr0
    if z0 > 0.5:
        z0 = 0.5
    if z0 < 0.01:
        z0 = 0.01

    if el < 0.0:
        xel = -el
        if xel <= xl(z0, -70.0, 4.35):
            lstab = 1
        elif xel <= xl(z0, -85.2, 0.502):
            lstab = 2
        elif xel <= xl(z0, -245.0, 0.050):
            lstab = 3
        else:
            lstab = 4
    else:
        if el >= xl(z0, -327.0, 0.627):
            lstab = 4
        elif el >= xl(z0, -70.0, 0.295):
            lstab = 5
        else:
            lstab = 6

    return lstab

def distf(state: AermodState) -> None:
    """SUBROUTINE DISTF (calc2.f)."""
    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        state.XMAX = (
            state.UP
            * math.atan2(state.FM * state.BVPRIM, -state.FB)
            / state.BVPRIM
        )

        state.DHFAER = (
            2.66 * (state.FB / (state.BVF * state.BVF * state.UP)) ** state.THIRD
        )
        xln = state.FB / (state.UP * state.USTAR * state.USTAR)
        delhnn = 1.2 * xln**0.6 * (state.HSP + 1.2 * xln) ** 0.4
        state.DHFAER = min(state.DHFAER, delhnn)

        # Compute Neutral/Unstable Final Rise
        if state.FB <= 0.0:
            xmaxn = (
                4.0
                * state.DS
                * (state.VS + 3.0 * state.UP)
                * (state.VS + 3.0 * state.UP)
                / (state.VS * state.UP)
            )
            state.DHFAER = min(
                state.DHFAER, 3.0 * state.DS * state.VS / state.UP
            )
        else:
            if state.FB >= 55.0:
                xmaxn = 119.0 * state.FB**0.4
            else:
                xmaxn = 49.0 * state.FB**0.625
            cblprd(state, xmaxn)
            state.DHFAER = min(state.DHFAER, state.DHP1)

        # Apply calm, stable rise limit
        state.DHFAER = min(
            state.DHFAER,
            4.0 * state.FB**0.25 / (state.BVF * state.BVF) ** 0.375,
        )

        dhfsav = state.DHFAER

        if state.URBSTAB:
            if state.HSP + state.DHFAER >= state.ZI and state.HSP < state.ZI:
                bvzi2 = (state.G / state.PTATZI) * 0.005
                state.PSUBS = state.FB / (
                    state.UP
                    * bvzi2
                    * (state.ZI - state.HSP)
                    * (state.ZI - state.HSP)
                    * (state.ZI - state.HSP)
                )
                state.HEDHH = (17.576 * state.PSUBS + 0.296296) ** state.THIRD

                if state.HEDHH < (2.0 * state.THIRD):
                    state.PPF = 0.0
                elif state.HEDHH > 2.0:
                    state.PPF = 1.0
                else:
                    state.PPF = 1.5 - (1.0 / state.HEDHH)

                if state.PPF > 0.0:
                    if state.PPF == 1.0:
                        state.DHFAER = state.HEDHH * (state.ZI - state.HSP)
                    else:
                        state.DHFAER = (
                            0.75 * (state.ZI - state.HSP) * state.HEDHH
                            + 0.5 * (state.ZI - state.HSP)
                        )
                else:
                    state.DHFAER = dhfsav
    else:
        # Unstable plume
        if state.FB <= 0.0:
            state.XMAX = (
                4.0
                * state.DS
                * (state.VS + 3.0 * state.UP)
                * (state.VS + 3.0 * state.UP)
                / (state.VS * state.UP)
            )
            state.DHFAER = 3.0 * state.DS * state.VS / state.UP
        else:
            if state.FB >= 55.0:
                state.XMAX = 119.0 * state.FB**0.4
            else:
                state.XMAX = 49.0 * state.FB**0.625
            cblprd(state, state.XMAX)
            state.DHFAER = state.DHP1


def wakflg(state: AermodState) -> None:
    """SUBROUTINE WAKFLG (calc2.f)."""
    if state.IFVSEC <= state.NSEC and state.NSEC > 0:
        # Fortran (IFVSEC, ISRC) -> [state.IFVSEC - 1, state.ISRC - 1]
        sec_idx = state.IFVSEC - 1

        state.DSBH = float(state.ADSBH[sec_idx, src_idx])
        state.DSBW = float(state.ADSBW[sec_idx, src_idx])

        state.XADJ = float(state.ADSXADJ[sec_idx, src_idx])

        state.RSCALE = (

        )

def xydist(state: AermodState, indx: int) -> None:
    """SUBROUTINE XYDIST (calc2.f)."""
    # Fortran INDX (1-based) -> idx = indx - 1
    idx = indx - 1
    state.XR = float(state.AXR[idx])
    state.YR = float(state.AYR[idx])
    state.ZELEV = float(state.AZELEV[idx])
    state.ZHILL = float(state.AZHILL[idx])
    state.ZFLAG = float(state.AZFLAG[idx])

    # Calculate Downwind (X) and Crosswind (Y) Distances
    state.X = -(
        (state.XR - state.XS) * state.WDSIN
        + (state.YR - state.YS) * state.WDCOS
    )
    state.Y = (state.XR - state.XS) * state.WDCOS - (
        state.YR - state.YS
    ) * state.WDSIN

    # Calculate Source-Receptor (Radial) Distance, DISTR
    state.DISTR = math.sqrt(state.X * state.X + state.Y * state.Y)

    # Calculate height of receptor above stack base, ZRT
    # Fortran L_FLATSRC(ISRC) -> state.L_FLATSRC[state.ISRC - 1]
    if state.L_FLATSRC[state.ISRC - 1]:
        state.ZRT = state.ZFLAG
    else:
        state.ZRT = state.ZELEV - state.ZS + state.ZFLAG

    if state.SCREEN:
        state.X = state.DISTR
        state.Y = 0.0

def meandr(state: AermodState, uef: float, svef: float) -> float:
    """SUBROUTINE MEANDR (calc2.f)."""
    tran = state.BIGT * 3600.0

    sqrtarg = uef * uef - 2.0 * svef * svef
    if sqrtarg >= 0.01:
        umean = math.sqrt(sqrtarg)
    else:
        umean = 0.1

    totkin = uef * uef
    ttrav = state.DISTR / uef
    sigran = 2.0 * svef * svef + umean * umean * (1.0 - math.exp(-ttrav / tran))
    fran = sigran / totkin

    if state.L_PBal:
        fran = math.sqrt(fran)

    fran = min(fran, state.FRANMAX)
    fran = max(fran, state.FRANMIN)

    if state.DEBUG:
        isrc = state.ISRC
        irec = state.IREC
        comp = state.debug_comp
        # WRITE(DBGUNT,*) SVEF, UEF, UMEAN / DISTR, TTRAV, TRAN / TOTKIN, SIGRAN, FRAN
        state.emit_debug(isrc, irec, comp, "MEANDR", "SVEF", svef)
        state.emit_debug(isrc, irec, comp, "MEANDR", "UEF", uef)
        state.emit_debug(isrc, irec, comp, "MEANDR", "UMEAN", umean)
        state.emit_debug(isrc, irec, comp, "MEANDR", "DISTR", state.DISTR)
        state.emit_debug(isrc, irec, comp, "MEANDR", "TTRAV", ttrav)
        state.emit_debug(isrc, irec, comp, "MEANDR", "TRAN", tran)
        state.emit_debug(isrc, irec, comp, "MEANDR", "TOTKIN", totkin)
        state.emit_debug(isrc, irec, comp, "MEANDR", "SIGRAN", sigran)
        state.emit_debug(isrc, irec, comp, "MEANDR", "FRAN", fran)

    return fran

        # Fortran NL = 1..NLEV-1 -> nl in range(nlev - 1)
        for nl in range(nlev - 1):
            if srctyp in ("RLINE", "RLINEXT"):
                ws_nl = float(state.RL_GRIDWS[nl, state.I_ALPHA - 1])
                ls[nl] = 0.5 * ws_nl * ws_nl
            else:
                ws_nl = float(state.GRIDWS[nl])
                ls[nl] = 0.5 * ws_nl * ws_nl

        # Fortran LS(NLEV) -> ls[nlev - 1]
        ls[nlev - 1] = 0.5 * uathh * uathh
        rs[nlev - 1] = 0.0

        # Fortran NL = NLEV-1, 1, -1 -> nl_1b in range(nlev - 1, 0, -1)
        for nl_1b in range(nlev - 1, 0, -1):
            nl = nl_1b - 1
            if nl_1b < nlev - 1:
                zmid = 0.5 * (
                    float(state.GRIDHT[nl + 1]) + float(state.GRIDHT[nl])
                )
                pthc = 0.5 * (
                    float(state.GRIDPT[nl + 1]) + float(state.GRIDPT[nl])
                )
                tghc = 0.5 * (
                    float(state.GRIDTG[nl + 1]) + float(state.GRIDTG[nl])
                )
                topht = float(state.GRIDHT[nl + 1])
            elif nl_1b == nlev - 1:
                zmid = 0.5 * (hhill + float(state.GRIDHT[nl]))
                pthc = 0.5 * (ptathh + float(state.GRIDPT[nl]))
                tghc = 0.5 * (tgathh + float(state.GRIDTG[nl]))
                topht = hhill

            xn2[nl] = (state.G / pthc) * tghc
            rs[nl] = float(rs[nl + 1]) + float(xn2[nl]) * (
                (hhill - zmid) * (topht - float(state.GRIDHT[nl]))
            )

        it = 1
        # Fortran NL = NLEV, 1, -1 -> nl_1b in range(nlev, 0, -1)
        for nl_1b in range(nlev, 0, -1):
            if ls[nl_1b - 1] >= rs[nl_1b - 1]:
                it = nl_1b

        if it > 1:
            if it == nlev:
                wstop = uathh
                htop = hhill
            else:
                # Fortran IT -> it - 1
                if srctyp in ("RLINE", "RLINEXT"):
                    wstop = float(state.RL_GRIDWS[it - 1, state.I_ALPHA - 1])
                    htop = float(state.GRIDHT[it - 1])
                else:
                    wstop = float(state.GRIDWS[it - 1])
                    htop = float(state.GRIDHT[it - 1])

            ib = it - 1
            # Fortran IB -> ib - 1
            hbot = float(state.GRIDHT[ib - 1])
            if srctyp in ("RLINE", "RLINEXT"):
                dws = (
                    wstop - float(state.RL_GRIDWS[ib - 1, state.I_ALPHA - 1])
                ) / (htop - hbot)
            else:
                dws = (wstop - float(state.GRIDWS[ib - 1])) / (htop - hbot)

            dws2 = dws * dws
            n2 = float(xn2[ib - 1])

            a = 0.5 * (n2 - dws2)
            b = htop * dws2 - wstop * dws - n2 * hhill
            c = (
                (n2 * hhill * htop)
                - 0.5 * (n2 * htop * htop)
                - 0.5 * (dws2 * htop * htop)
                + wstop * dws * htop
                - (float(ls[it - 1]) - float(rs[it - 1]))
            )
            b2 = b * b
            ac4 = 4.0 * a * c
            if -0.001 < (b2 - ac4) / b2 < 0.0:
                ac4 = b2
            deter = math.sqrt(b2 - ac4)
            state.HCRIT = (-b - deter) / (2.0 * a)
        else:
            state.HCRIT = 0.0
    else:
        state.HCRIT = 0.0


def pdf(state: AermodState) -> None:
    """SUBROUTINE PDF (calc2.f)."""
    skcalc(state)
    crcalc(state)
    alcalc(state)
    becalc(state)
    aacalc(state)
    bbcalc(state)
    llcalc(state)


def skcalc(state: AermodState) -> None:
    """SUBROUTINE SKCALC (calc2.f)."""
    if state.SURFAC:
        wbar3 = 1.25 * (state.WSTAR**3) * (state.CENTER / state.ZI)
    else:
        wbar3 = 0.125 * (state.WSTAR**3)

    state.SKEW = wbar3 / (state.SWEFFD * state.SWEFFD * state.SWEFFD)


def crcalc(state: AermodState) -> None:
    """SUBROUTINE CRCALC (calc2.f)."""
    state.R = 2.0


def alcalc(state: AermodState) -> None:
    """SUBROUTINE ALCALC (calc2.f)."""
    state.ALPHPD = (1.0 + state.R * state.R) / (1.0 + 3.0 * state.R * state.R)


def becalc(state: AermodState) -> None:
    """SUBROUTINE BECALC (calc2.f)."""
    state.BETAPD = 1.0 + state.R * state.R


def aacalc(state: AermodState) -> None:
    """SUBROUTINE AACALC (calc2.f)."""
    determ = (state.ALPHPD * state.ALPHPD) * (state.SKEW * state.SKEW) + (
        4.0 / state.BETAPD
    )
    swratio = state.SWEFFD / state.WSTAR
    state.ASUB1 = swratio * (
        0.5 * state.ALPHPD * state.SKEW + 0.5 * math.sqrt(determ)
    )
    state.ASUB2 = swratio * (
        0.5 * state.ALPHPD * state.SKEW - 0.5 * math.sqrt(determ)
    )


def bbcalc(state: AermodState) -> None:
    """SUBROUTINE BBCALC (calc2.f)."""
    state.BSUB1 = state.R * state.ASUB1
    state.BSUB2 = -state.R * state.ASUB2


def llcalc(state: AermodState) -> None:
    """SUBROUTINE LLCALC (calc2.f)."""
    state.LAMDA1 = state.ASUB2 / (state.ASUB2 - state.ASUB1)
    state.LAMDA2 = 1.0 - state.LAMDA1


def decay(state: AermodState, xarg: float) -> None:
    """SUBROUTINE DECAY (calc2.f)."""
    state.D = 1.0
    dc1 = state.DECOEF

    # Fortran ISRC -> state.ISRC - 1, ICSTAT(7) -> state.ICSTAT[6], ICSTAT(8) -> state.ICSTAT[7]
    isrc_idx = state.ISRC - 1
    urbsrc_flag = str(state.URBSRC[isrc_idx]).strip()
    pollut = state.POLLUT.strip()

    if (
        state.URBAN
        and pollut == "SO2"
        and urbsrc_flag == "Y"
        and ((state.ICSTAT[6] == 0 and state.ICSTAT[7] == 0) or state.DFAULT)
    ):
        state.DECOEF = 4.81e-5
    elif (pollut == "SO2" and urbsrc_flag == "N") or state.DFAULT:
        state.DECOEF = 0.0

    if state.DECOEF > 0.0:
        if state.STABLE or (state.UNSTAB and state.HS >= state.ZI):
            state.D = math.exp(
                max(state.EXPLIM, -state.DECOEF * xarg / state.UEFF)
            )
        else:
            state.D = math.exp(
                max(state.EXPLIM, -state.DECOEF * xarg / state.UEFFD)
            )

    state.DECOEF = dc1


def vrtsbl(
    state: AermodState, szarg: float, hearg: float, ziarg: float
) -> None:
    """SUBROUTINE VRTSBL (calc2.f)."""
    v = 0.0

    if state.ZR == 0.0:
        a1 = (-0.5 / (szarg * szarg)) * hearg * hearg
        if a1 > state.EXPLIM:
            v = math.exp(a1)
        sum_val = 0.0
        # Fortran DO I = 1, 100 -> i in range(1, 101)
        for i in range(1, 101):
            t = 0.0
            twoizi = 2.0 * float(i) * ziarg
            a2 = (-0.5 / (szarg * szarg)) * (twoizi - hearg) * (twoizi - hearg)
            a3 = (-0.5 / (szarg * szarg)) * (twoizi + hearg) * (twoizi + hearg)
            if a2 > state.EXPLIM:
                t = math.exp(a2)
            if a3 > state.EXPLIM:
                t = t + math.exp(a3)
            sum_val = sum_val + t

            if abs(t) <= 5.0e-7 * abs(sum_val):
                break
        v = 2.0 * (v + sum_val)
    elif state.ZR <= ziarg:
        hetmp = max(hearg, hearg - state.ZR)
        zitmp = max(ziarg, ziarg - state.ZR)

        a1 = (-0.5 / (szarg * szarg)) * (state.ZR - hetmp) * (state.ZR - hetmp)
        a2 = (-0.5 / (szarg * szarg)) * (state.ZR + hetmp) * (state.ZR + hetmp)
        if a1 > state.EXPLIM:
            v = math.exp(a1)
        if a2 > state.EXPLIM:
            v = v + math.exp(a2)
        sum_val = 0.0
        # Fortran DO I = 1, 100 -> i in range(1, 101)
        for i in range(1, 101):
            t = 0.0
            twoizi = 2.0 * float(i) * zitmp
            a3 = (
                (-0.5 / (szarg * szarg))
                * (state.ZR - (twoizi - hetmp))
                * (state.ZR - (twoizi - hetmp))
            )
            a4 = (
                (-0.5 / (szarg * szarg))
                * (state.ZR + (twoizi - hetmp))
                * (state.ZR + (twoizi - hetmp))
            )
            a5 = (
                (-0.5 / (szarg * szarg))
                * (state.ZR - (twoizi + hetmp))
                * (state.ZR - (twoizi + hetmp))
            )
            a6 = (
                (-0.5 / (szarg * szarg))
                * (state.ZR + (twoizi + hetmp))
                * (state.ZR + (twoizi + hetmp))
            )
            if a3 > state.EXPLIM:
                t = t + math.exp(a3)
            if a4 > state.EXPLIM:
                t = t + math.exp(a4)
            if a5 > state.EXPLIM:
                t = t + math.exp(a5)
            if a6 > state.EXPLIM:
                t = t + math.exp(a6)
            sum_val = sum_val + t

            if abs(t) <= 1.0e-6 * abs(sum_val):
                break
        v = v + sum_val
    else:
        v = 0.0

    state.FSUBZ = v / (state.SRT2PI * szarg)


def vrtsbn(state: AermodState, szarg: float, hearg: float) -> None:
    """SUBROUTINE VRTSBN (calc2.f)."""
    v = 0.0

    if state.ZR == 0.0:
        a1 = (-0.5 / (szarg * szarg)) * hearg * hearg
        if a1 > state.EXPLIM:
            v = math.exp(a1)
        v = 2.0 * v
    else:
        hetmp = max(hearg, hearg - state.ZR)
        a1 = (-0.5 / (szarg * szarg)) * (state.ZR - hetmp) * (state.ZR - hetmp)
        a2 = (-0.5 / (szarg * szarg)) * (state.ZR + hetmp) * (state.ZR + hetmp)
        if a1 > state.EXPLIM:
            v = math.exp(a1)
        if a2 > state.EXPLIM:
            v = v + math.exp(a2)

    state.FSUBZ = v / (state.SRT2PI * szarg)


def vrtbl_cbl(
    state: AermodState,
    he1: float,
    he2: float,
    sz1: float,
    sz2: float,
    fact: float,
) -> None:
    """SUBROUTINE VRTCBL (calc2.f)."""
    v = 0.0

    if abs(state.ZR - 0.0) < 1.0e-10:
        sum_val = 0.0
        # Fortran DO I = 0, 1000 -> i in range(0, 1001)
        for i in range(0, 1001):
            t1 = 0.0
            t2 = 0.0
            twoizi = 2.0 * float(i) * state.ZI * fact
            if fact < 0.0 and i == 0:
                continue

            hearg1 = twoizi + he1
            hearg2 = twoizi + he2
            a1 = (-0.5 / (sz1 * sz1)) * hearg1 * hearg1
            if a1 > state.EXPLIM:
                t1 = math.exp(a1)
            a2 = (-0.5 / (sz2 * sz2)) * hearg2 * hearg2
            if a2 > state.EXPLIM:
                t2 = math.exp(a2)

            term = (state.LAMDA1 / sz1) * t1 + (state.LAMDA2 / sz2) * t2
            sum_val = sum_val + term

            if abs(term) <= 5.0e-7 * abs(sum_val):
                break
        v = 2.0 * sum_val
    elif state.ZR <= state.ZI:
        he1tmp = max(he1, he1 - state.ZR)
        he2tmp = max(he2, he2 - state.ZR)
        zitmp = max(state.ZI, state.ZI - state.ZR)

        sum_val = 0.0
        # Fortran DO I = 0, 1000 -> i in range(0, 1001)
        for i in range(0, 1001):
            t1 = 0.0
            t2 = 0.0
            twoizi = 2.0 * float(i) * zitmp * fact
            if fact < 0.0 and i == 0:
                continue

            hearg1 = twoizi + he1tmp
            hearg2 = twoizi + he2tmp
            a1 = (
                (-0.5 / (sz1 * sz1))
                * (state.ZR - hearg1)
                * (state.ZR - hearg1)
            )
            a2 = (
                (-0.5 / (sz1 * sz1))
                * (state.ZR + hearg1)
                * (state.ZR + hearg1)
            )
            if a1 > state.EXPLIM:
                t1 = math.exp(a1)
            if a2 > state.EXPLIM:
                t1 = t1 + math.exp(a2)
            a3 = (
                (-0.5 / (sz2 * sz2))
                * (state.ZR - hearg2)
                * (state.ZR - hearg2)
            )
            a4 = (
                (-0.5 / (sz2 * sz2))
                * (state.ZR + hearg2)
                * (state.ZR + hearg2)
            )
            if a3 > state.EXPLIM:
                t2 = math.exp(a3)
            if a4 > state.EXPLIM:
                t2 = t2 + math.exp(a4)

            term = (state.LAMDA1 / sz1) * t1 + (state.LAMDA2 / sz2) * t2
            sum_val = sum_val + term

            if abs(term) <= 1.0e-6 * abs(sum_val):
                break
        v = sum_val
    else:
        v = 0.0

    state.FSUBZ = v / state.SRT2PI


def pfract(state: AermodState, hearg: float) -> None:
    """SUBROUTINE PFRACT (calc2.f)."""
    state.PHEE = 0.0

    if state.STABLE and (state.HCRIT > 0.0):
        hcint = min(state.HSBL, state.HCRIT)

        a1 = (hcint - hearg) / (state.RTOF2 * state.SZ)
        a2 = (hcint + hearg) / (state.RTOF2 * state.SZ)
        b1 = erfx(a1)
        b2 = erfx(a2)

        sum_val = 0.0
        # Fortran DO I = 1, 100 -> i in range(1, 101)
        for i in range(1, 101):
            t = 0.0
            twoizi = 2.0 * float(i) * state.HSBL
            a3 = (hcint - hearg + twoizi) / (state.RTOF2 * state.SZ)
            a4 = (hcint + hearg + twoizi) / (state.RTOF2 * state.SZ)
            a5 = (hcint - hearg - twoizi) / (state.RTOF2 * state.SZ)
            a6 = (hcint + hearg - twoizi) / (state.RTOF2 * state.SZ)
            b3 = erfx(a3)
            b4 = erfx(a4)
            b5 = erfx(a5)
            b6 = erfx(a6)

            t = b3 + b4 + b5 + b6
            sum_val = sum_val + t

            if abs(t) <= 1.0e-6 * abs(sum_val):
                if i >= 5:
                    break

        state.PHEE = 0.5 * (b1 + b2 + sum_val)

        if state.PHEE > 1.01 and not state.L_SkipMessages:
            state.PHEE = 1.0


def erfx(arg: float) -> float:
    """FUNCTION ERFX (calc2.f)."""
    if arg > 4.0:
        res = 1.0
    elif arg < -4.0:
        res = -1.0
    elif abs(arg) < 1.0e-10:
        res = 0.0
    else:
        x = abs(arg)
        res = (
            1.0
            - 1.0
            / (
                1.0
                + x
                * (
                    0.705230784e-1
                    + x
                    * (
                        0.422820123e-1
                        + x
                        * (
                            0.92705272e-2
                            + x
                            * (
                                0.1520143e-3
                                + x * (0.2765672e-3 + x * 0.430638e-4)
                            )
                        )
                    )
                )
            )
            ** 16.0
        )
        if arg < 0.0:
            res = -res
    return res

"""SUBROUTINE DELTAH, HSPRIM, SBLRIS, CBLPRD, CBLPRN, CBLPR3 (prise.f)."""

from __future__ import annotations

import math

from .state import AermodState
from .iblval import locate
from .metext import gintrp


def deltah(state: AermodState, xarg: float) -> None:
    """SUBROUTINE DELTAH (prise.f)."""
    if (state.STABLE or (state.UNSTAB and (state.HS >= state.ZI))) and (
        xarg >= state.XMAX
    ):
        state.DHP = state.DHFAER

    elif (state.STABLE or (state.UNSTAB and (state.HS >= state.ZI))) and (
        xarg < state.XMAX
    ):
        xmaxtmp = (
            state.UP
            * math.atan2(state.FM * state.BVPRIM, -state.FB)
            / state.BVPRIM
        )
        xrise = min(xarg, xmaxtmp)
        sblris(state, xrise)
        kiter = 0

        isrc = state.ISRC
        irec = state.IREC
        comp = state.debug_comp

        while True:
            zplm = state.HSP + 0.5 * state.DHP
            dhpold = state.DHP

            # Locate index below ZPLM
            ndxzpl = locate(state.GRIDHT, 1, state.MXGLVL, zplm)
            # Fortran NDXZPL -> idx = ndxzpl - 1, NDXZPL+1 -> idx + 1
            idx = ndxzpl - 1
            ht_lo = float(state.GRIDHT[idx])
            ht_hi = float(state.GRIDHT[idx + 1])

            svpm = gintrp(
                ht_lo,
                float(state.GRIDSV[idx]),
                ht_hi,
                float(state.GRIDSV[idx + 1]),
                zplm,
            )
            upm = gintrp(
                ht_lo,
                float(state.GRIDWS[idx]),
                ht_hi,
                float(state.GRIDWS[idx + 1]),
                zplm,
            )

            svpm = max(svpm, state.SVMIN, state.SVUMIN * upm)
            if state.L_VECTORWS:
                upm = math.sqrt(upm * upm + 2.0 * svpm * svpm)
            upm = max(upm, state.WSMIN)

            state.UP = 0.5 * (state.US + upm)

            tgpm = gintrp(
                ht_lo,
                float(state.GRIDTG[idx]),
                ht_hi,
                float(state.GRIDTG[idx + 1]),
                zplm,
            )
            ptpm = gintrp(
                ht_lo,
                float(state.GRIDPT[idx]),
                ht_hi,
                float(state.GRIDPT[idx + 1]),
                zplm,
            )

            state.TGP = 0.5 * (state.TGS + tgpm)
            ptp = 0.5 * (state.PTS + ptpm)
            state.BVF = math.sqrt(state.G * state.TGP / ptp)
            if state.BVF < 1.0e-10:
                state.BVF = 1.0e-10
            state.BVPRIM = 0.7 * state.BVF

            xmaxtmp = (
                state.UP
                * math.atan2(state.FM * state.BVPRIM, -state.FB)
                / state.BVPRIM
            )
            xrise = min(xarg, xmaxtmp)
            sblris(state, xrise)

            kiter = kiter + 1

            if state.DEBUG:
                # WRITE(DBGUNT,6001) KITER, DHPOLD, DHP, ZPLM, UP, TGP
                state.emit_debug(isrc, irec, comp, "DELTAH", "DHPOLD", dhpold)
                state.emit_debug(isrc, irec, comp, "DELTAH", "DHP", state.DHP)
                state.emit_debug(isrc, irec, comp, "DELTAH", "ZPLM", zplm)
                state.emit_debug(isrc, irec, comp, "DELTAH", "UP", state.UP)
                state.emit_debug(isrc, irec, comp, "DELTAH", "TGP", state.TGP)

            # Check for convergence
            if (
                state.DHP > 0.0
                and abs((dhpold - state.DHP) / state.DHP) < 0.001
                and kiter >= 5
            ):
                if state.DHP <= 1.0e-5:
                    state.DHP = 1.0e-5
                break
            elif kiter < 10:
                continue

            if kiter >= 5:
                state.DHP = 0.5 * (state.DHP + dhpold)
                break

        # After completing iteration, reset UP and TGP to stack top values
        state.UP = state.US
        state.TGP = state.TGS
        ptp = state.PTS
        state.BVF = math.sqrt(state.G * state.TGP / ptp)
        if state.BVF < 1.0e-10:
            state.BVF = 1.0e-10
        state.BVPRIM = 0.7 * state.BVF

        # Make sure SBL rise is not greater than CBL rise
        cblprd(state, xarg)
        state.DHP = min(state.DHP, state.DHP1)
        state.DHP = min(state.DHP, state.DHFAER)

    elif state.UNSTAB:
        # Compute plume rise for the direct plume --- CALL CBLPRD
        cblprd(state, xarg)

        # Compute plume rise for the indirect plume --- CALL CBLPRN
        cblprn(state, xarg)

        if state.PPF > 0.0:
            cblpr3(state)
        else:
            state.DHP3 = 0.0


def hsprim(us: float, vs: float, hs: float, ds: float) -> float:
    """FUNCTION HSPRIM (prise.f)."""
    if vs < 1.5 * us:
        res = hs - 2.0 * ds * (1.5 - vs / us)
    else:
        res = hs

    if res < 0.0:
        res = 0.0

    return res


def sblris(state: AermodState, xarg: float) -> None:
    """SUBROUTINE SBLRIS (prise.f)."""
    terma = state.FB / (state.BVF * state.BVF * state.UP)
    termb = state.BVPRIM * state.FM / state.FB
    termc = math.sin(state.BVPRIM * xarg / state.UP)
    termd = math.cos(state.BVPRIM * xarg / state.UP)

    terme = termb * termc + 1.0 - termd

    if terme > 0.0:
        state.DHP = 2.66 * (terma * (termb * termc + 1.0 - termd)) ** state.THIRD
    else:
        state.DHP = 2.66 * max(0.0, terma * termb * termc) ** state.THIRD

    xln = state.FB / (state.UP * state.USTAR * state.USTAR)
    delhnn = 1.2 * xln**0.6 * (state.HSP + 1.2 * xln) ** 0.4

    state.DHP = min(state.DHP, state.DHFAER, delhnn)


def cblprd(state: AermodState, xarg: float) -> None:
    """SUBROUTINE CBLPRD (prise.f)."""
    state.DHP1 = (
        3.0 * state.FM * xarg / (state.BETA1 * state.BETA1 * state.UP * state.UP)
        + 3.0
        * state.FB
        * xarg
        * xarg
        / (2.0 * state.BETA1 * state.BETA1 * state.UP * state.UP * state.UP)
    ) ** state.THIRD


def cblprn(state: AermodState, xarg: float) -> None:
    """SUBROUTINE CBLPRN (prise.f)."""
    rsubh = state.BETA2 * (state.ZI - state.HSP)
    ryrz = (
        rsubh * rsubh
        + 0.25
        * state.ASUBE
        * (state.LAMDAY**1.5)
        * state.WSTAR
        * state.WSTAR
        * xarg
        * xarg
        / (state.UP * state.UP)
    )
    delhi = (
        math.sqrt((2.0 * state.FB * state.ZI) / (state.ALPHAR * state.UP * ryrz))
        * (xarg / state.UP)
    )
    state.DHP2 = delhi


def cblpr3(state: AermodState) -> None:
    """SUBROUTINE CBLPR3 (prise.f)."""
    if state.PPF == 1.0:
        state.DHP3 = state.HEDHH * (state.ZI - state.HSP)
    else:
        state.DHP3 = (
            0.75 * (state.ZI - state.HSP) * state.HEDHH
            + 0.5 * (state.ZI - state.HSP)
        )

"""SUBROUTINE SIGY, SIGZ, BID, SZSFCL, RMSSIG (sigmas.f)."""

from __future__ import annotations

import math

from .state import AermodState
from .iblval import locate
from .metext import gintrp


def sigy(state: AermodState, xarg: float) -> None:
    """SUBROUTINE SIGY (sigmas.f)."""
    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        svovru = max(state.SVUMIN, state.SVEFF / state.UEFF)
        tyeff = (state.ZIMECH / (156.0 * state.SVEFF)) * (
            max(state.HE, 0.46) / 0.46
        )
        state.SYAMB = (svovru * xarg) / (
            1.0 + xarg / (2.0 * state.UEFF * tyeff)
        ) ** 0.3

    elif state.UNSTAB:
        svovru = max(state.SVUMIN, state.SVEFFD / state.UEFFD)
        xnodim = svovru * xarg / state.ZI
        denomi = max(state.HS, 0.46)
        betalph = max(78.0 * (0.46 / denomi), 0.7)
        state.SYAMB = (svovru * xarg) / (1.0 + betalph * xnodim) ** 0.3

        state.SYAN = state.SYAMB

        if state.PPF > 0.0:
            svovru = max(state.SVUMIN, state.SVEFF3 / state.UEFF3)
            tyeff3 = (state.ZIMECH / (156.0 * state.SVEFF3)) * (
                max(state.HE3, 0.46) / 0.46
            )
            state.SYA3 = (svovru * xarg) / (
                1.0 + xarg / (2.0 * state.UEFF3 * tyeff3)
            ) ** 0.3
        else:
            state.SYA3 = 0.0


def sigz(state: AermodState, xarg: float) -> None:
    """SUBROUTINE SIGZ (sigmas.f)."""
    if state.STABLE or (state.UNSTAB and state.HS >= state.ZI):
        ttravl = xarg / state.UEFF

        ndxzhe = locate(state.GRIDHT, 1, state.MXGLVL, state.HE)
        # Fortran NDXZHE -> idx = ndxzhe - 1, NDXZHE+1 -> idx + 1
        idx = ndxzhe - 1
        ptp = gintrp(
            float(state.GRIDHT[idx]),
            float(state.GRIDPT[idx]),
            float(state.GRIDHT[idx + 1]),
            float(state.GRIDPT[idx + 1]),
            state.HE,
        )

        if state.TGEFF > 0.0:
            bvfrq = math.sqrt(state.G * state.TGEFF / ptp)
        else:
            bvfrq = 1.0e-10

        if bvfrq < 1.0e-10:
            bvfrq = 1.0e-10

        ztmp = max(state.HS, state.HE, 1.0e-4)

        if state.URBSTAB:
            state.SZAMB = (
                state.SWEFF
                * ttravl
                / math.sqrt(
                    1.0 + state.SWEFF * ttravl * (1.0 / (0.72 * ztmp))
                )
            )
        else:
            state.SZAMB = (
                state.SWEFF
                * ttravl
                / math.sqrt(
                    1.0
                    + state.SWEFF
                    * ttravl
                    * (1.0 / (0.72 * ztmp) + bvfrq / (0.54 * state.SWEFF))
                )
            )

        if state.HE < state.ZI:
            szsfcl(state, xarg)
            sigf = min(state.HE / state.ZI, 1.0)
            state.SZAS = (1.0 - sigf) * state.SZSURF + sigf * state.SZAMB
        else:
            state.SZAS = state.SZAMB

    elif state.UNSTAB:
        if state.PPF < 1.0:
            if not state.SURFAC:
                alphab = 1.0
            else:
                alphab = 0.6 + 0.4 * (state.CENTER / (0.1 * state.ZI))

            state.SZAD1 = alphab * state.BSUB1 * state.WSTAR * xarg / state.UEFFD
            state.SZAD2 = alphab * state.BSUB2 * state.WSTAR * xarg / state.UEFFD

            szsfcl(state, xarg)

            state.SZAD1 = math.sqrt(
                state.SZAD1 * state.SZAD1 + state.SZSURF * state.SZSURF
            )
            state.SZAD2 = math.sqrt(
                state.SZAD2 * state.SZAD2 + state.SZSURF * state.SZSURF
            )
        else:
            state.SZAD1 = 1.0
            state.SZAD2 = 1.0

        state.SZAN1 = state.SZAD1
        state.SZAN2 = state.SZAD2

        if state.PPF > 0.0:
            ttravl = xarg / state.UEFF3
            state.SZA3 = (
                state.SWEFF3
                * ttravl
                / math.sqrt(
                    1.0 + state.SWEFF3 * ttravl * (1.0 / (0.72 * state.HE3))
                )
            )
        else:
            state.SZA3 = 0.0


def bid(state: AermodState) -> None:
    """SUBROUTINE BID (sigmas.f)."""
    # Fortran ISRC -> state.ISRC - 1
    aftsrc_flag = str(state.AFTSRC[state.ISRC - 1]).strip()

    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        if aftsrc_flag == "Y":
            state.SZB = state.PLUME_SPREAD / 5.93
            state.SYB = state.PLUME_SPREAD / 2.28
            state.SZBD = 0.0
            state.SZBN = 0.0
            state.SYB3 = 0.0
            state.SZB3 = 0.0
        else:
            state.SZB = state.BETA2 * state.DHP / state.RTOF2
            state.SYB = state.SZB
            state.SZBD = 0.0
            state.SZBN = 0.0
            state.SYB3 = 0.0
            state.SZB3 = 0.0
    elif state.UNSTAB:
        if aftsrc_flag == "Y":
            state.SZBD = state.PLUME_SPREAD / 5.93
            state.SZBN = state.SZBD
        else:
            state.SZBD = state.BETA2 * state.DHP1 / state.RTOF2
            state.SZBN = state.SZBD

        if state.PPF > 0.0:
            state.SZB3 = state.BETA2 * state.PPF * state.DHP3 / state.RTOF2
        else:
            state.SZB3 = 0.0

        if aftsrc_flag == "Y":
            state.SYB = 2.6 * state.SZBD
            state.SYB3 = state.SZB3
        else:
            state.SYB = state.SZBD
            state.SYB3 = state.SZB3


def szsfcl(state: AermodState, xarg: float) -> None:
    """SUBROUTINE SZSFCL (sigmas.f)."""
    if state.UNSTAB and state.SURFAC:
        state.SZSURF = (
            state.BSUBC
            * (1.0 - 10.0 * (state.CENTER / state.ZI))
            * (state.USTAR / state.UEFFD)
            * (state.USTAR / state.UEFFD)
            * (xarg * xarg / abs(state.OBULEN))
        )
    elif state.STABLE:
        state.SZSURF = (
            (state.RTOF2 / state.RTOFPI)
            * state.USTAR
            * (xarg / state.UEFF)
            * (1.0 + 0.7 * xarg / state.OBULEN) ** (-1.0 * state.THIRD)
        )
    else:
        state.SZSURF = 0.0


def rmssig(state: AermodState) -> None:
    """SUBROUTINE RMSSIG (sigmas.f)."""
    if state.STABLE or (state.UNSTAB and (state.HS >= state.ZI)):
        state.SY = math.sqrt(
            state.SYB * state.SYB
            + state.SYAMB * state.SYAMB
            + state.VSIGY * state.VSIGY
            + state.PLATSY * state.PLATSY
        )
        state.SZ = math.sqrt(
            state.SZB * state.SZB
            + state.SZAS * state.SZAS
            + state.VSIGZ * state.VSIGZ
            + state.PLATSZ * state.PLATSZ
        )

        state.SYN = 0.0
        state.SZD1 = 0.0
        state.SZD2 = 0.0
        state.SZN1 = 0.0
        state.SZN2 = 0.0
        state.SY3 = 0.0
        state.SZ3 = 0.0

    elif state.UNSTAB:
        max_platsyd = max(state.PLATSYD1, state.PLATSYD2)
        max_platsyn = max(state.PLATSYN1, state.PLATSYN2)
        state.SY = math.sqrt(
            state.SYB * state.SYB
            + state.SYAMB * state.SYAMB
            + state.VSIGY * state.VSIGY
            + max_platsyd * max_platsyd
        )
        state.SYN = math.sqrt(
            state.SYB * state.SYB
            + state.SYAN * state.SYAN
            + state.VSYN * state.VSYN
            + max_platsyn * max_platsyn
        )
        state.SZD1 = math.sqrt(
            state.SZBD * state.SZBD
            + state.SZAD1 * state.SZAD1
            + state.VSZD1 * state.VSZD1
            + state.PLATSZD1 * state.PLATSZD1
        )
        state.SZD2 = math.sqrt(
            state.SZBD * state.SZBD
            + state.SZAD2 * state.SZAD2
            + state.VSZD2 * state.VSZD2
            + state.PLATSZD2 * state.PLATSZD2
        )
        state.SZN1 = math.sqrt(
            state.SZBN * state.SZBN
            + state.SZAN1 * state.SZAN1
            + state.VSZN1 * state.VSZN1
            + state.PLATSZN1 * state.PLATSZN1
        )
        state.SZN2 = math.sqrt(
            state.SZBN * state.SZBN
            + state.SZAN2 * state.SZAN2
            + state.VSZN2 * state.VSZN2
            + state.PLATSZN2 * state.PLATSZN2
        )

        if state.PPF > 0.0:
            state.SY3 = math.sqrt(
                state.SYB3 * state.SYB3
                + state.SYA3 * state.SYA3
                + state.PLATSYP * state.PLATSYP
            )
            state.SZ3 = math.sqrt(
                state.SZB3 * state.SZB3
                + state.SZA3 * state.SZA3
                + state.PLATSZP * state.PLATSZP
            )
        else:
            state.SY3 = 0.0
            state.SZ3 = 0.0
