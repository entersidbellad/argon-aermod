=== HYPOTHESIS ===
`aermod_py.hrloop` and `aermod_py.metext` have a circular top-level import (`hrloop` imports `metext` from `.metext` before defining `cent_date` and `julian`, while `metext` imports `cent_date, julian` from `.hrloop`). Loading the package submodules in `aermod_py/__init__.py` with a temporary module `__getattr__` that executes unstarted modules on demand and returns a forwarding proxy for attributes on partially initialized modules resolves the circular import cleanly.
=== FILE: aermod_py/__init__.py ===
"""AERMOD 26135 Python port: package entry point."""

from __future__ import annotations

import importlib.util
import pathlib
import sys

import numpy as xp
import glue


def _load_package():
    """Load package submodules while resolving circular top-level imports."""
    pkg_dir = pathlib.Path(__file__).resolve().parent
    pkg_mod = sys.modules[__name__]
    stems = [
        p.stem
        for p in sorted(pkg_dir.glob("*.py"))
        if p.stem != "__init__" and not p.stem.startswith("_")
    ]
    specs = {}
    mods = {}
    started = set()
    finished = set()

    for stem in stems:
        full_name = f"{__name__}.{stem}"
        if full_name in sys.modules:
            mods[stem] = sys.modules[full_name]
            started.add(stem)
            finished.add(stem)
            continue
        spec = importlib.util.find_spec(full_name)
        if spec is None or spec.loader is None:
            continue
        mod = importlib.util.module_from_spec(spec)
        specs[stem] = spec
        mods[stem] = mod

        def _make_getattr(s=stem, m=mod, sp=spec, fn=full_name):
            def __getattr__(attr: str):
                if attr.startswith("_"):
                    raise AttributeError(attr)
                if s not in started:
                    started.add(s)
                    sp.loader.exec_module(m)
                    finished.add(s)
                if attr in m.__dict__:
                    return m.__dict__[attr]
                if s not in finished:
                    def _proxy(*args, **kwargs):
                        return m.__dict__[attr](*args, **kwargs)
                    return _proxy
                raise AttributeError(f"module {fn!r} has no attribute {attr!r}")

            return __getattr__

        mod.__dict__["__getattr__"] = _make_getattr()
        sys.modules[full_name] = mod
        setattr(pkg_mod, stem, mod)

    order = [s for s in ("state", "hrloop") if s in specs] + [
        s for s in stems if s in specs and s not in ("state", "hrloop")
    ]
    for stem in order:
        if stem not in started:
            started.add(stem)
            specs[stem].loader.exec_module(mods[stem])
            finished.add(stem)

    for mod in mods.values():
        mod.__dict__.pop("__getattr__", None)

    return mods["state"].AermodState, mods["hrloop"].hrloop


AermodState, hrloop = _load_package()


def run_case(
    project: glue.Project,
    met: glue.Met,
    setup: dict,
    *,
    debug: bool = False,
) -> glue.Result:
    """Run HRLOOP and all in-scope AERMOD physics for a prepared case."""
    state = AermodState.from_setup(project, met, setup, debug=debug)
    hrloop(state)
    dates = xp.asarray(state.dates_out, dtype=xp.int64)
    if len(state.conc_out) > 0:
        conc = xp.stack(state.conc_out, axis=0).astype(xp.float64)
    else:
        conc = xp.zeros((0, state.NUMREC), dtype=xp.float64)
    return glue.Result(dates=dates, conc=conc, debug=state.debug_rows)
